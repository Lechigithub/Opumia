// Kit for "FTX" v30, art direction "The bet" (2026-11-24), written from nothing: no component, colour, typeface or
// motion of an earlier film or version. An empire built like a house of cards on belief, which the narration ends by
// calling a bet. Felt green and cream like a card table, chip coral for what is lost; one heavy extra-condensed
// typeface (Sofia Sans Extra Condensed 900) carved with depth like the channel's slabs, and one display serif (DM Serif
// Display) for what people said. Words arrive as whole phrases and stay (owner: "never rushed, so viewers can actually
// read the word"); the spoken word lights up inside its phrase instead of popping in on its own.
import React from "react";
import { AbsoluteFill, Easing, interpolate, useCurrentFrame, useVideoConfig } from "remotion";
import { loadFont as sofia } from "@remotion/google-fonts/SofiaSansExtraCondensed";
import { loadFont as dmserif } from "@remotion/google-fonts/DMSerifDisplay";
const SLAB = sofia("normal", { weights: ["900"], subsets: ["latin"] }).fontFamily;
const SERIF = dmserif("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
const FELT = "#0f3d2e", FELT_DK = "#06221a", CREAM = "#f6ecd6", CORAL = "#ff5b3a", INK = "#13110e";
type P = Record<string, any>;
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const out = Easing.bezier(0.16, 1, 0.3, 1);
const useW = (p: P) => { const { width } = useVideoConfig(); return Number(p.boxW ?? width); };
/** carved depth: a stack of offset shadows (the channel's slab type has thickness), then a soft drop */
const depth = (c: string, n: number, px: number) => Array.from({ length: n }, (_, i) => `${(i + 1) * px * 0.5}px ${(i + 1) * px}px 0 ${c}`).join(", ") + `, ${n * px}px ${n * px * 2}px ${n * px * 3}px rgba(0,0,0,.55)`;

/** A whole phrase arrives as one block (a slow rise out of a soft blur) and stays; each word brightens when it is said. */
const Phrase: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const words: Array<{ text: string; from: number; c?: string }> = p.words ?? [];
  const lines: typeof words[] = [[]];
  for (const w of words) { if (w.text === "|") lines.push([]); else lines.at(-1)!.push(w); }
  const first = Math.min(...words.filter((w) => w.text !== "|").map((w) => w.from));
  // the phrase leaves (a slow fade and drift) after its hold, so a long panel does not carry words the whole time
  const gone = p.until !== undefined ? interpolate(f, [Number(p.until), Number(p.until) + 12], [1, 0], clamp) : 1;
  const k = interpolate(f, [first - 4, first + 14], [0, 1], { ...clamp, easing: out }) * gone;
  const longest = Math.max(...lines.map((l) => l.reduce((a, w) => a + w.text.length + 1, 0)));
  const size = Number(p.size ?? Math.min(W * 0.22, (W * 0.98) / (longest * 0.36)));
  const al = String(p.align ?? "center");
  const serif = p.font === "serif";
  return <div style={{ width: "100%", display: "flex", flexDirection: "column", alignItems: al === "left" ? "flex-start" : al === "right" ? "flex-end" : "center", opacity: k, transform: `translateY(${(1 - k) * size * 0.5}px)`, filter: k < 0.99 ? `blur(${(1 - k) * 10}px)` : undefined }}>
    {lines.map((l, i) => <div key={i} style={{ display: "flex", gap: size * (serif ? 0.24 : 0.16), flexWrap: "wrap", justifyContent: al === "left" ? "flex-start" : "center", lineHeight: serif ? 1.1 : 0.92 }}>{l.map((w, j) => {
      const lit = interpolate(f, [w.from - 2, w.from + 6], [0, 1], clamp);
      const hot = w.c === "r" ? CORAL : CREAM;
      const col = w.c ? (lit > 0.5 ? hot : CREAM) : CREAM; const dim = 0.82 + 0.18 * lit;
      const b = p.build ? interpolate(f, [w.from - 3, w.from + 7], [0, 1], { ...clamp, easing: out }) : 1;
      return <span key={j} style={{ display: "inline-block", transform: p.build ? `translateY(${(1 - b) * size * 0.4}px) scale(${0.9 + 0.1 * b})` : undefined, ...(p.build ? { filter: b < 0.99 ? `blur(${(1 - b) * 6}px)` : undefined } : {}), fontFamily: serif ? SERIF : SLAB, fontWeight: serif ? 400 : 900, fontSize: serif ? size * 0.8 : size, textTransform: serif ? "none" : "uppercase", letterSpacing: serif ? 0 : size * 0.005, color: col, opacity: dim * (p.build ? interpolate(f, [w.from - 3, w.from + 5], [0, 1], clamp) : 1), textShadow: serif ? "0 6px 30px rgba(0,0,0,.85)" : depth(w.c === "r" && lit > 0.5 ? "#8a2112" : FELT_DK, 5, size * 0.012) }}>{w.text}</span>;
    })}</div>)}
  </div>;
};

/** A figure set like a chip value: the whole number at once, carved deep, its label in serif above. Never ticks through
 *  numbers nobody said. */
const Figure: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const text = String(p.text ?? "");
  const size = Number(p.size ?? Math.min(W * 0.5, (W * 0.98) / (text.length * 0.42)));
  const k = interpolate(f, [0, 16], [0, 1], { ...clamp, easing: out });
  const c = p.fill === "red" ? CORAL : CREAM;
  return <div style={{ width: "100%", textAlign: "center" }}>
    {p.label ? <div style={{ fontFamily: SERIF, fontSize: size * 0.17, color: CREAM, opacity: interpolate(f, [6, 18], [0, 1], clamp), textShadow: "0 4px 18px rgba(0,0,0,.9)" }}>{String(p.label)}</div> : null}
    <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: size, lineHeight: 0.9, color: c, opacity: k, transform: `scale(${1.25 - 0.25 * k})`, filter: k < 0.99 ? `blur(${(1 - k) * 8}px)` : undefined, textShadow: depth(p.fill === "red" ? "#7d1d0f" : "#3c3426", 7, size * 0.008) }}>{text}</div>
  </div>;
};

"flex-start" : p.align === "right" ? "flex-end" : "center", transform: `scale(${grow}) rotate(${p.tilt ?? 0}deg)`, transformOrigin: "50% 50%", opacity: gone, filter: `${GLOW}${gone < 1 ? ` blur(${(1 - gone) * 14}px)` : ""}` }}>

// ── a general performed beat: one or two plates (the second dissolving in on a word), a hero cut-out, words, rain
const Beat: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { width: W, height: H, fps } = useVideoConfig(); const len = Number(p.len);
  const two = p.second as { slot: number; at: number; trim?: number; push?: [number, number]; grade?: string } | undefined;
  const k2 = two ? interpolate(f, [two.at * fps - 10, two.at * fps + 4], [0, 1], clamp) : 0;
  const hero = p.hero as { slot: number; x: number; w: number; at?: number; light?: string } | undefined;
  const rain = p.rain as { kind: string; n: number; from?: number } | undefined;
  return <AbsoluteFill style={{ background: "#000", overflow: "hidden" }}>
    <Plate src={S(p, Number(p.bg ?? 0))} trim={Number(p.trim ?? 0)} len={len} push={(p.push as [number, number]) ?? [1.05, 1.2]} pan={(p.pan as [number, number]) ?? [0.02, -0.02]} bloom={Number(p.bloom ?? 0.36)} {...(p.grade ? { grade: String(p.grade) } : {})} />
    {two && k2 > 0 ? <AbsoluteFill style={{ opacity: k2 }}><Plate src={S(p, two.slot)} trim={two.trim ?? 0} from={Math.max(0, two.at - 0.5)} len={len} push={two.push ?? [1.1, 1.25]} bloom={0.45} {...(two.grade ? { grade: two.grade } : {})} /></AbsoluteFill> : null}
    {rain ? <AbsoluteFill style={{ perspective: W * 0.9 }}><DepthRain n={rain.n} t={f - (rain.from ?? 0) * fps} layer="far" render={(i) => rain.kind === "coupon" ? <div style={{ transform: "scale(1)" }}><Coupon text="FTT" small="issued by FTX" stub="FTX" boxW={W * 0.16} rot={0} /></div> : <TwoSided front={S(p, 8)!} back={S(p, 9)!} w={W * 0.16} />} /></AbsoluteFill> : null}
    <Light hot={Number(p.hot ?? 0.32)} />
    {hero ? <HeroCutout src={S(p, hero.slot)!} x={hero.x} w={hero.w} light={hero.light ?? "rgba(255,150,50,.75)"} core="rgba(255,225,170,.95)" fadeTop={0.1} scale={interpolate(f, [0, len * fps], [1.02, 1.09])} opacity={interpolate(f, [((hero.at ?? 0) * fps) - 8, (hero.at ?? 0) * fps + 4], [0, 1], clamp)} extraFilter="saturate(1.15) contrast(1.08)" /> : null}
    {rain ? <AbsoluteFill style={{ perspective: W * 0.9 }}><DepthRain n={rain.n} t={f - (rain.from ?? 0) * fps} layer="near" render={() => rain.kind === "coupon" ? <Coupon text="FTT" small="issued by FTX" stub="FTX" boxW={W * 0.16} rot={0} /> : <TwoSided front={S(p, 8)!} back={S(p, 9)!} w={W * 0.16} />} /></AbsoluteFill> : null}
    {((p.words ?? []) as Array<React.ComponentProps<typeof Words>>).map((w, i) => <Words key={i} {...w} size={w.size * W} />)}
    <Flash len={len} inF={Number(p.inF ?? 7)} outF={Number(p.outF ?? 4)} at={(p.flashes as number[]) ?? []} />
  </AbsoluteFill>;
};
