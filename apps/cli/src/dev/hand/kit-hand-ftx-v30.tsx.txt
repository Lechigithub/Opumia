// Kit for "FTX" v30, art direction "The bet" (2026-11-24), written from nothing: no component, colour, typeface or
// motion of an earlier film or version. An empire built like a house of cards on belief, which the narration ends by
// calling a bet. Felt green and cream like a card table, chip coral for what is lost; one heavy extra-condensed
// typeface (Sofia Sans Extra Condensed 900) carved with depth like the channel's slabs, and one display serif (DM Serif
// Display) for what people said. Words arrive as whole phrases and stay (owner: "never rushed, so viewers can actually
// read the word"); the spoken word lights up inside its phrase instead of popping in on its own.
import React from "react";
import { AbsoluteFill, Easing, interpolate, useCurrentFrame, useVideoConfig } from "remotion";
import { loadFont as sofia } from "@remotion/google-fonts/SofiaSansExtraCondensed";
import { loadFont as dmserif } from "@remotion/google-fonts/DMSerifDisplay";
const SLAB = sofia("normal", { weights: ["900"], subsets: ["latin"] }).fontFamily;
const SERIF = dmserif("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
const FELT = "#0f3d2e", FELT_DK = "#06221a", CREAM = "#f6ecd6", CORAL = "#ff5b3a", INK = "#13110e";
type P = Record<string, any>;
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const out = Easing.bezier(0.16, 1, 0.3, 1);
const useW = (p: P) => { const { width } = useVideoConfig(); return Number(p.boxW ?? width); };
/** carved depth: a stack of offset shadows (the channel's slab type has thickness), then a soft drop */
const depth = (c: string, n: number, px: number) => Array.from({ length: n }, (_, i) => `${(i + 1) * px * 0.5}px ${(i + 1) * px}px 0 ${c}`).join(", ") + `, ${n * px}px ${n * px * 2}px ${n * px * 3}px rgba(0,0,0,.55)`;

/** A whole phrase arrives as one block (a slow rise out of a soft blur) and stays; each word brightens when it is said. */
const Phrase: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const words: Array<{ text: string; from: number; c?: string }> = p.words ?? [];
  const lines: typeof words[] = [[]];
  for (const w of words) { if (w.text === "|") lines.push([]); else lines.at(-1)!.push(w); }
  const first = Math.min(...words.filter((w) => w.text !== "|").map((w) => w.from));
  // the phrase leaves (a slow fade and drift) after its hold, so a long panel does not carry words the whole time
  const gone = p.until !== undefined ? interpolate(f, [Number(p.until), Number(p.until) + 12], [1, 0], clamp) : 1;
  const k = interpolate(f, [first - 4, first + 14], [0, 1], { ...clamp, easing: out }) * gone;
  const longest = Math.max(...lines.map((l) => l.reduce((a, w) => a + w.text.length + 1, 0)));
  const size = Number(p.size ?? Math.min(W * 0.22, (W * 0.98) / (longest * 0.36)));
  const al = String(p.align ?? "center");
  const serif = p.font === "serif";
  return <div style={{ width: "100%", display: "flex", flexDirection: "column", alignItems: al === "left" ? "flex-start" : al === "right" ? "flex-end" : "center", opacity: k, transform: `translateY(${(1 - k) * size * 0.5}px)`, filter: k < 0.99 ? `blur(${(1 - k) * 10}px)` : undefined }}>
    {lines.map((l, i) => <div key={i} style={{ display: "flex", gap: size * (serif ? 0.24 : 0.16), flexWrap: "wrap", justifyContent: al === "left" ? "flex-start" : "center", lineHeight: serif ? 1.1 : 0.92 }}>{l.map((w, j) => {
      const lit = interpolate(f, [w.from - 2, w.from + 6], [0, 1], clamp);
      const hot = w.c === "r" ? CORAL : CREAM;
      const col = w.c ? (lit > 0.5 ? hot : CREAM) : CREAM; const dim = 0.82 + 0.18 * lit;
      const b = p.build ? interpolate(f, [w.from - 3, w.from + 7], [0, 1], { ...clamp, easing: out }) : 1;
      return <span key={j} style={{ display: "inline-block", transform: p.build ? `translateY(${(1 - b) * size * 0.4}px) scale(${0.9 + 0.1 * b})` : undefined, ...(p.build ? { filter: b < 0.99 ? `blur(${(1 - b) * 6}px)` : undefined } : {}), fontFamily: serif ? SERIF : SLAB, fontWeight: serif ? 400 : 900, fontSize: serif ? size * 0.8 : size, textTransform: serif ? "none" : "uppercase", letterSpacing: serif ? 0 : size * 0.005, color: col, opacity: dim * (p.build ? interpolate(f, [w.from - 3, w.from + 5], [0, 1], clamp) : 1), textShadow: serif ? "0 6px 30px rgba(0,0,0,.85)" : depth(w.c === "r" && lit > 0.5 ? "#8a2112" : FELT_DK, 5, size * 0.012) }}>{w.text}</span>;
    })}</div>)}
  </div>;
};

/** A figure set like a chip value: the whole number at once, carved deep, its label in serif above. Never ticks through
 *  numbers nobody said. */
const Figure: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const text = String(p.text ?? "");
  const size = Number(p.size ?? Math.min(W * 0.5, (W * 0.98) / (text.length * 0.42)));
  const k = interpolate(f, [0, 16], [0, 1], { ...clamp, easing: out });
  const c = p.fill === "red" ? CORAL : CREAM;
  return <div style={{ width: "100%", textAlign: "center" }}>
    {p.label ? <div style={{ fontFamily: SERIF, fontSize: size * 0.17, color: CREAM, opacity: interpolate(f, [6, 18], [0, 1], clamp), textShadow: "0 4px 18px rgba(0,0,0,.9)" }}>{String(p.label)}</div> : null}
    <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: size, lineHeight: 0.9, color: c, opacity: k, transform: `scale(${1.25 - 0.25 * k})`, filter: k < 0.99 ? `blur(${(1 - k) * 8}px)` : undefined, textShadow: depth(p.fill === "red" ? "#7d1d0f" : "#3c3426", 7, size * 0.008) }}>{text}</div>
  </div>;
};

/** What a person said, in the display serif, with who said it. */
const Said: React.FC<P> = (p) => <div style={{ width: "100%" }}>
  <Phrase {...p} font="serif" align={p.align ?? "left"} />
  {p.by ? <SaidBy by={String(p.by)} from={Number(p.byFrom ?? 18)} size={Number(p.size ?? 120)} /> : null}
</div>;
const SaidBy: React.FC<{ by: string; from: number; size: number }> = ({ by, from, size }) => { const f = useCurrentFrame(); return <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: size * 0.26, letterSpacing: size * 0.02, color: CORAL, marginTop: size * 0.2, opacity: interpolate(f, [from, from + 10], [0, 1], clamp), textTransform: "uppercase" }}>{by}</div>; };

/** A name on a cream bar that slides out from behind its first letter; the role in serif under it. */
const Name: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const name = String(p.text ?? "");
  const size = Number(p.size ?? (W * 0.95) / (name.length * 0.4));
  const k = interpolate(f, [0, 14], [0, 1], { ...clamp, easing: out });
  return <div style={{ display: "inline-flex", flexDirection: "column", alignItems: "flex-start" }}>
    <div style={{ background: CREAM, padding: `${size * 0.04}px ${size * 0.14}px`, clipPath: `inset(0 ${(1 - k) * 100}% 0 0)`, boxShadow: `${size * 0.05}px ${size * 0.06}px 0 ${CORAL}` }}>
      <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: size, lineHeight: 1, color: INK, textTransform: "uppercase" }}>{name}</div>
    </div>
    {p.role ? <div style={{ fontFamily: SERIF, fontSize: size * 0.32, color: CREAM, marginTop: size * 0.1, opacity: interpolate(f, [10, 22], [0, 1], clamp), textShadow: "0 4px 18px rgba(0,0,0,.9)" }}>{String(p.role)}</div> : null}
  </div>;
};

/** A date or place as a casino chip rolling in: a coral disc with cream edge notches and the words round its face. */
const Chip: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const k = interpolate(f, [0, 18], [0, 1], { ...clamp, easing: out });
  const lines = String(p.text ?? "").split("|");
  return <div style={{ width: W, height: W, transform: `translateX(${(1 - k) * -W * 1.4}px) rotate(${(1 - k) * -300}deg)`, position: "relative" }}>
    <div style={{ position: "absolute", inset: 0, borderRadius: "50%", background: `repeating-conic-gradient(${CREAM} 0 9deg, ${CORAL} 9deg 36deg)`, boxShadow: `0 ${W * 0.05}px ${W * 0.12}px rgba(0,0,0,.6)` }} />
    <div style={{ position: "absolute", inset: W * 0.1, borderRadius: "50%", background: CORAL, border: `${W * 0.015}px dashed ${CREAM}`, display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center" }}>
      {lines.map((l, i) => <div key={i} style={{ fontFamily: SLAB, fontWeight: 900, fontSize: W * (i === 0 ? 0.2 : 0.12), lineHeight: 0.95, color: CREAM, textTransform: "uppercase" }}>{l}</div>)}
    </div>
  </div>;
};

/** His post, set by the kit from his own words (not a screenshot): a dark card with the handle and the lines. */
const Post: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const lines: string[] = p.lines ?? [];
  const k = interpolate(f, [0, 14], [0, 1], { ...clamp, easing: out });
  const fs = W * 0.078;
  return <div style={{ width: W, background: "#101312", borderRadius: W * 0.03, padding: W * 0.05, boxShadow: `0 ${W * 0.04}px ${W * 0.1}px rgba(0,0,0,.7)`, opacity: k, transform: `translateY(${(1 - k) * 60}px)` }}>
    <div style={{ display: "flex", alignItems: "center", gap: fs * 0.5, marginBottom: fs * 0.6 }}>
      <div style={{ width: fs * 1.6, height: fs * 1.6, borderRadius: "50%", background: CORAL }} />
      <div><div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: fs * 0.9, color: CREAM }}>{String(p.name ?? "")}</div><div style={{ fontFamily: SERIF, fontSize: fs * 0.6, color: "rgba(246,236,214,.6)" }}>{String(p.handle ?? "")}</div></div>
    </div>
    {lines.map((l, i) => { const a = interpolate(f, [3 + i * 4, 9 + i * 4], [0, 1], clamp); return <div key={i} style={{ fontFamily: SERIF, fontSize: fs, lineHeight: 1.25, color: CREAM, opacity: a, background: i >= Number(p.mark ?? 99) ? `linear-gradient(90deg, ${CORAL}55 ${a * 100}%, transparent ${a * 100}%)` : undefined }}>{l}</div>; })}
  </div>;
};

/** The leaked balance sheet, drawn from its verified lines (not a screenshot): a cream ledger, a row struck coral. */
const Ledger: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const rows: Array<{ a: string; b: string; mark?: boolean; from?: number }> = p.rows ?? [];
  const fs = W * 0.068;
  return <div style={{ width: W, background: CREAM, padding: W * 0.05, boxShadow: `0 ${W * 0.03}px ${W * 0.08}px rgba(0,0,0,.6)`, transform: `rotate(${p.rot ?? -2}deg)` }}>
    <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: fs * 1.6, color: INK, textTransform: "uppercase", borderBottom: `${fs * 0.12}px solid ${INK}`, paddingBottom: fs * 0.3, marginBottom: fs * 0.4 }}>{String(p.title ?? "")}</div>
    {rows.map((r, i) => { const m = r.mark ? interpolate(f, [r.from ?? 20, (r.from ?? 20) + 12], [0, 1], clamp) : 0; return <div key={i} style={{ display: "flex", justifyContent: "space-between", fontFamily: SERIF, fontSize: fs, color: INK, padding: `${fs * 0.25}px ${fs * 0.2}px`, borderBottom: `1px solid ${INK}33`, background: `linear-gradient(90deg, ${CORAL}aa ${m * 100}%, transparent ${m * 100}%)` }}><span>{r.a}</span><span style={{ fontFamily: SLAB, fontWeight: 900 }}>{r.b}</span></div>; })}
  </div>;
};

/** A share as two stacks of chips: the part in coral, the rest cream, growing to their heights. */
const Stacks: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const share = Number(p.share ?? 0.4);
  const n = 20, cw = W * 0.34, ch = W * 0.04;
  const stack = (count: number, color: string, delay: number) => Array.from({ length: count }, (_, i) => { const a = interpolate(f, [delay + i * 0.8, delay + i * 0.8 + 6], [0, 1], clamp); return <div key={i} style={{ width: cw, height: ch, borderRadius: "50%", background: color, border: `${ch * 0.18}px dashed ${color === CORAL ? CREAM : CORAL}`, marginTop: -ch * 0.45, opacity: a, transform: `translateY(${(1 - a) * -80}px)`, boxShadow: "0 4px 8px rgba(0,0,0,.4)" }} />; }).reverse();
  return <div style={{ width: W, display: "flex", justifyContent: "space-around", alignItems: "flex-end" }}>
    {[[Math.round(n * share), CORAL, String(p.labelA ?? "")], [n - Math.round(n * share), CREAM, String(p.labelB ?? "")]].map(([c, col, lab], k) => <div key={k} style={{ display: "flex", flexDirection: "column", alignItems: "center" }}>
      <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: W * 0.12, color: col as string, textShadow: depth(FELT_DK, 4, W * 0.002), marginBottom: W * 0.03 }}>{lab}</div>
      {stack(c as number, col as string, k * 8)}
    </div>)}
  </div>;
};

/** A coupon: perforated cream ticket with a coral stub, the words printed on it. */
const Coupon: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const k = interpolate(f, [0, 16], [0, 1], { ...clamp, easing: out });
  return <div style={{ width: W, height: W * 0.42, display: "flex", transform: `rotate(${(p.rot ?? -6) + (1 - k) * 20}deg) scale(${0.8 + 0.2 * k})`, opacity: k, filter: "drop-shadow(0 20px 30px rgba(0,0,0,.6))" }}>
    <div style={{ flex: 3, background: CREAM, display: "flex", flexDirection: "column", justifyContent: "center", paddingLeft: W * 0.06, borderRight: `${W * 0.008}px dashed ${INK}` }}>
      <div style={{ fontFamily: SERIF, fontSize: W * 0.05, color: INK }}>{String(p.small ?? "")}</div>
      <div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: W * 0.17, lineHeight: 0.9, color: INK, textTransform: "uppercase" }}>{String(p.text ?? "")}</div>
    </div>
    <div style={{ flex: 1, background: CORAL, display: "flex", alignItems: "center", justifyContent: "center" }}><div style={{ fontFamily: SLAB, fontWeight: 900, fontSize: W * 0.1, color: CREAM, transform: "rotate(-90deg)" }}>{String(p.stub ?? "")}</div></div>
  </div>;
};

/** The table: green felt with its weave and a pool of light. */
const Stage: React.FC<P> = () => {
  const f = useCurrentFrame(); const { width: W } = useVideoConfig();
  return <AbsoluteFill style={{ background: `radial-gradient(ellipse at ${50 + Math.sin(f / 90) * 6}% 42%, #1d6b4f 0%, ${FELT} 45%, ${FELT_DK} 100%)` }}>
    <AbsoluteFill style={{ opacity: 0.18, backgroundImage: `repeating-linear-gradient(45deg, rgba(0,0,0,.5) 0 1px, transparent 1px ${W / 400}px), repeating-linear-gradient(-45deg, rgba(255,255,255,.25) 0 1px, transparent 1px ${W / 380}px)` }} />
  </AbsoluteFill>;
};
const Chapter: React.FC<P> = (p) => <AbsoluteFill><Stage /><AbsoluteFill style={{ justifyContent: "center", alignItems: "center" }}><Phrase words={[{ text: String(p.title ?? ""), from: 0 }]} /></AbsoluteFill></AbsoluteFill>;
const Plain: React.FC<P> = (p) => <AbsoluteFill><Stage /><div style={{ position: "absolute", left: "10%", top: "12%", fontFamily: SLAB, fontWeight: 900, fontSize: 90, color: CREAM }}>{String(p.title ?? p.headline ?? "")}</div></AbsoluteFill>;

/** The cut, when there is one: a cream card swept across the frame like a dealt card. */
const Deal: React.FC<{ progress: number }> = ({ progress }) => {
  const x = interpolate(progress, [0, 1], [-130, 130]);
  return <AbsoluteFill style={{ pointerEvents: "none" }}><div style={{ position: "absolute", top: "-20%", bottom: "-20%", width: "60%", left: `${x}%`, transform: "rotate(12deg)", background: `linear-gradient(90deg, transparent, ${CREAM}cc 40%, ${CREAM}cc 60%, transparent)`, opacity: Math.sin(progress * Math.PI) }} /></AbsoluteFill>;
};
