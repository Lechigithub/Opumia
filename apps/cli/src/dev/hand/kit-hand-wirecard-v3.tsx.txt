// Kit for "Wirecard" (hand film 2, 2026-11-25), written from nothing: no component, colour, typeface or motion of the
// FTX film. Concept "Numbers on glass": an empire of payments that existed as numbers on screens. Cold steel and
// ice-white light, electric cyan while it rises, warning red when it falls. One tall condensed face (League Gothic) for
// the line's key words, filled ice-white to cyan, glowing; small labels in Fjalla One. Built for MOTION (owner: close
// the channel's motion gap): every frame keeps travelling — data and money streaming through depth, plates panning
// hard, whip pans between pictures, words slamming in and shoving each other out, a counter running.
import React from "react";
import { AbsoluteFill, Easing, Img, interpolate, staticFile, useCurrentFrame, useVideoConfig } from "remotion";
import { loadFont as gothicF } from "@remotion/google-fonts/LeagueGothic";
import { loadFont as fjallaF } from "@remotion/google-fonts/FjallaOne";
import { BuildWords, DepthRain, FlashCut, GradedPlate, LightLayer, LightWash, Whip, rnd, shake, EASE_OUT } from "@craft";
const GOTHIC = gothicF("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
const FJALLA = fjallaF("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
type P = Record<string, any>;
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const ICE = "#eaf6ff", CYAN = "#3fe0ff", STEEL = "#0a1420", RED = "#ff2d3d";
const FILL_ICE = "linear-gradient(180deg, #ffffff 0%, #dff4ff 40%, #7fe6ff 100%)";
const FILL_RED = "linear-gradient(180deg, #ffd6d6 0%, #ff5a5a 50%, #c5101e 100%)";
const GLOW_ICE = "drop-shadow(0 0 12px rgba(63,224,255,.75)) drop-shadow(0 0 34px rgba(30,140,255,.4)) drop-shadow(0 8px 10px rgba(0,0,0,.7))";
const GLOW_RED = "drop-shadow(0 0 12px rgba(255,45,61,.8)) drop-shadow(0 0 34px rgba(255,20,40,.4)) drop-shadow(0 8px 10px rgba(0,0,0,.7))";
/** the film's grade: cool steel shadows, neutral highlights, contrast high */
// v2: deeper shadows, highlights left saturated (the channel: gold against deep darks; 13 % of frames under luma 0.12)
const STEELG = "saturate(1.05) contrast(1.42) brightness(0.84) hue-rotate(4deg)";
const S = (p: P, k: number) => (p.slots ?? {})[String(k)] as string | undefined;

type Wd = { text: string; at: number; br?: boolean; c?: "r" | "y" };
type WSpec = { words: Wd[]; x: number; y: number; w: number; size: number; style?: "ice" | "red" | "label"; until?: number; align?: "left" | "center" | "right"; enter?: "rise" | "fly" | "spin"; slam?: boolean };

/** The film's words: League Gothic in ice or red; labels in Fjalla One. A slammed block lands with a jolt. */
const Words: React.FC<WSpec> = (p) => {
  const f = useCurrentFrame(); const { width: W, fps } = useVideoConfig();
  const first = Math.min(...p.words.map((w) => w.at));
  const jolt = p.slam ? shake(f, Math.round(first * fps) + 4, 7, 16) : 0;
  const label = p.style === "label";
  return <div style={{ position: "absolute", inset: 0, transform: `translate(${jolt}px, ${jolt * 0.6}px)` }}>
    <BuildWords words={p.words} x={p.x} y={p.y} w={p.w} size={p.size * W} fontFamily={label ? FJALLA : GOTHIC} fill={p.style === "red" ? FILL_RED : label ? "linear-gradient(#cfe9ff, #cfe9ff)" : FILL_ICE} hi={p.style === "red" ? FILL_ICE : FILL_RED} glow={p.style === "red" ? GLOW_RED : GLOW_ICE} charEm={label ? 0.5 : 0.36} enter={p.enter ?? (p.slam ? "fly" : "rise")} {...(p.until !== undefined ? { until: p.until } : {})} {...(p.align ? { align: p.align } : {})} />
  </div>;
};

/** Numbers streaming through depth: euro signs and figures flying past the camera (the empire of numbers). */
const DataRain: React.FC<{ n: number; from: number; colour: string; speed?: number }> = ({ n, from, colour, speed = 0.024 }) => {
  const f = useCurrentFrame(); const { width: W } = useVideoConfig(); const { fps } = useVideoConfig();
  if (f < from * fps) return null;
  const glyphs = ["€", "1.9", "24", "€", "0", "%", "€", "9", "DAX", "€"];
  return <AbsoluteFill style={{ perspective: W * 0.7, mixBlendMode: "screen", pointerEvents: "none" }}>
    <DepthRain n={n} t={f - from * fps} speed={speed} zFrom={-1800} zTo={600} spread={2.6} render={(i) => <div style={{ fontFamily: GOTHIC, fontSize: W * (0.04 + rnd(i, 11) * 0.09), color: colour, textShadow: `0 0 18px ${colour}`, opacity: 0.55 + rnd(i, 12) * 0.45 }}>{glyphs[i % glyphs.length]}</div>} />
  </AbsoluteFill>;
};

/** A figure counting up to its value, fast at first, landing on the spoken word. */
const Counter: React.FC<{ to: number; at: number; dur: number; prefix?: string; suffix?: string; x: number; y: number; size: number; style?: "ice" | "red" }> = (p) => {
  const f = useCurrentFrame(); const { width: W, height: H, fps } = useVideoConfig();
  if (f < (p.at - 0.4) * fps) return null;
  const k = interpolate(f, [(p.at - 0.4) * fps, (p.at - 0.4 + p.dur) * fps], [0, 1], { ...clamp, easing: Easing.out(Easing.cubic) });
  const v = Math.round(p.to * k);
  const s = interpolate(f, [(p.at - 0.4) * fps, (p.at - 0.4 + 0.3) * fps], [2.2, 1], { ...clamp, easing: EASE_OUT });
  return <div style={{ position: "absolute", left: 0, right: 0, top: H * p.y, display: "flex", justifyContent: "center", transform: `scale(${s})`, filter: p.style === "red" ? GLOW_RED : GLOW_ICE }}>
    <span style={{ fontFamily: GOTHIC, fontSize: W * p.size, backgroundImage: p.style === "red" ? FILL_RED : FILL_ICE, WebkitBackgroundClip: "text", WebkitTextFillColor: "transparent", color: "transparent", fontVariantNumeric: "tabular-nums" }}>{p.prefix ?? ""}{v}{p.suffix ?? ""}</span>
  </div>;
};

/** One word shoves another out of the frame: the old one is struck sideways and spins away, the new one slams in. */
const Shove: React.FC<{ out: string; inn: string; at: number; y: number; size: number }> = (p) => {
  const f = useCurrentFrame(); const { width: W, height: H, fps } = useVideoConfig();
  const k = interpolate(f, [p.at * fps - 2, p.at * fps + 9], [0, 1], { ...clamp, easing: Easing.in(Easing.cubic) });
  const kin = interpolate(f, [p.at * fps - 4, p.at * fps + 3], [0, 1], { ...clamp, easing: Easing.out(Easing.cubic) });
  const word = (t: string, fill: string, glow: string, style: React.CSSProperties) => <div style={{ position: "absolute", left: 0, right: 0, top: H * p.y, textAlign: "center", fontFamily: GOTHIC, fontSize: W * p.size, whiteSpace: "nowrap", ...style }}><span style={{ backgroundImage: fill, WebkitBackgroundClip: "text", WebkitTextFillColor: "transparent", color: "transparent", filter: glow }}>{t}</span></div>;
  return <>
    {k < 1 ? word(p.out, "linear-gradient(#b9c6d3, #7d8b99)", "drop-shadow(0 8px 10px rgba(0,0,0,.7))", { transform: `translateX(${k * W * 1.1}px) rotate(${k * 38}deg)`, opacity: 1 - k * 0.6, filter: k > 0.05 ? `blur(${k * 14}px)` : undefined }) : null}
    {kin > 0 ? word(p.inn, FILL_ICE, GLOW_ICE, { transform: `translateX(${(1 - kin) * -W * 1.1}px) scale(${1 + (1 - kin) * 0.4})`, filter: kin < 0.98 ? `blur(${(1 - kin) * 16}px)` : undefined }) : null}
  </>;
};

/** Police light thrown across the frame (red and blue sweeping). */
const Strobe: React.FC<{ from: number; to: number }> = ({ from, to }) => {
  const f = useCurrentFrame(); const { fps } = useVideoConfig();
  if (f < from * fps || f > to * fps) return null;
  const ph = Math.sin(f * 1.3), x = 50 + 40 * Math.sin(f * 0.5);
  return <AbsoluteFill style={{ mixBlendMode: "screen", background: `radial-gradient(ellipse at ${x}% 40%, ${ph > 0 ? "rgba(255,30,50,.45)" : "rgba(40,90,255,.45)"} 0%, rgba(0,0,0,0) 60%)` }} />;
};

/**
 * One beat of the film: plates that travel (each can arrive on a word and pan hard), a whip pan in or out, the film's
 * light, data or money streaming through depth, words, a counter, a shove, police light, and flashes.
 */
const Beat: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { width: W, fps } = useVideoConfig(); const len = Number(p.len);
  const plates = (p.plates ?? []) as Array<{ slot: number; trim?: number; at?: number; push?: [number, number]; pan?: [number, number]; rot?: [number, number]; grade?: string; whip?: boolean }>;
  const wIn = p.whipIn ? interpolate(f, [0, 7], [1, 0], { ...clamp, easing: Easing.out(Easing.cubic) }) : 0;
  const wOut = p.whipOut ? interpolate(f, [len * fps - 6, len * fps], [0, 1], { ...clamp, easing: Easing.in(Easing.cubic) }) : 0;
  const body = <AbsoluteFill style={{ background: STEEL, overflow: "hidden" }}>
    {plates.map((pl, i) => {
      const from = pl.at ?? 0;
      // a later plate arrives by a hard horizontal whip inside the beat
      const k = pl.at ? interpolate(f, [from * fps - 6, from * fps + 2], [0, 1], { ...clamp, easing: Easing.out(Easing.cubic) }) : 1;
      if (k <= 0) return null;
      return <AbsoluteFill key={i} style={pl.at ? { transform: `translateX(${(1 - k) * W * (i % 2 ? -1 : 1)}px)`, filter: k < 0.98 ? `blur(${(1 - k) * 22}px)` : undefined } : undefined}>
        <GradedPlate src={S(p, pl.slot)} trim={pl.trim ?? 0} from={Math.max(0, from - 0.3)} len={len - from} push={[(pl.push ?? [1.08, 1.45])[0], (pl.push ?? [1.08, 1.45])[0] + ((pl.push ?? [1.08, 1.45])[1] - (pl.push ?? [1.08, 1.45])[0]) * 1.5]} pan={pl.pan ?? [0.04, -0.04]} rot={pl.rot ?? [-1.5, 1.5]} bloom={0.3} grade={pl.grade ?? STEELG} />
      </AbsoluteFill>;
    })}
    {S(p, 7) ? <LightLayer src={S(p, 7)} opacity={Number(p.lightOpacity ?? 0.4)} trim={Number(p.lightTrim ?? 0)} filter={String(p.lightFilter ?? "hue-rotate(160deg) saturate(1.3)")} /> : null}
    <LightWash wash={p.red ? "#ff2d3d" : "#1e8cff"} washOpacity={0.16} hot={p.red ? "rgba(255,120,120,.18)" : "rgba(160,230,255,.18)"} edge="rgba(0,2,6,.94)" />
    {p.lowkey ? <AbsoluteFill style={{ background: "rgba(0,2,6,.5)", mixBlendMode: "multiply" }} /> : null}
    {p.data !== false ? <DataRain n={Number(p.data?.n ?? 64)} from={Number(p.data?.from ?? 0)} colour={p.red ? RED : CYAN} speed={Number(p.data?.speed ?? 0.026)} /> : null}
    {p.money ? <AbsoluteFill style={{ perspective: W * 0.8 }}><DepthRain n={Number(p.money.n ?? 26) * 1.6} t={f - Number(p.money.from ?? 0) * fps} speed={Number(p.money.speed ?? 0.01) * 1.8} render={(i) => S(p, 8) ? <Img src={staticFile(S(p, 8)!)} style={{ width: W * (0.1 + rnd(i, 21) * 0.06), boxShadow: "0 10px 24px rgba(0,0,0,.55)", filter: `brightness(${0.75 + rnd(i, 22) * 0.35})` }} /> : <div />} /></AbsoluteFill> : null}
    {p.strobe ? <Strobe from={Number(p.strobe[0])} to={Number(p.strobe[1])} /> : null}
    {p.shove ? <Shove out={p.shove.out} inn={p.shove.inn} at={Number(p.shove.at)} y={Number(p.shove.y ?? 0.34)} size={Number(p.shove.size ?? 0.2)} /> : null}
    {p.counter ? <Counter {...p.counter} /> : null}
    {((p.words ?? []) as WSpec[]).map((w, i) => <Words key={i} {...w} />)}
    <FlashCut len={len} inF={Number(p.inF ?? 0)} outF={Number(p.outF ?? 0)} at={(p.flashes as number[]) ?? []} colour="radial-gradient(ellipse at 50% 50%, #ffffff 0%, #d8f3ff 60%, #8fdcff 100%)" />
  </AbsoluteFill>;
  // a whip pan carries the beat in from, or out to, its neighbour
  return wIn > 0 ? <Whip k={wIn} dir="in">{body}</Whip> : wOut > 0 ? <Whip k={wOut} dir="out">{body}</Whip> : body;
};

/** The title card: the name on black, the light of the money behind it coming up. */
const Title: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { width: W, fps } = useVideoConfig();
  const len = Number(p.len ?? 2.4);
  const k = interpolate(f, [2, 14], [0, 1], { ...clamp, easing: EASE_OUT });
  return <AbsoluteFill style={{ background: "#02060b", justifyContent: "center", alignItems: "center", opacity: interpolate(f, [len * fps - 8, len * fps], [1, 0], clamp) }}>
    {p.data !== false ? <DataRain n={22} from={0} colour="#1d5f80" speed={0.006} /> : null}
    <div style={{ fontFamily: GOTHIC, fontSize: W * 0.19, letterSpacing: W * 0.012, backgroundImage: FILL_ICE, WebkitBackgroundClip: "text", WebkitTextFillColor: "transparent", filter: `${GLOW_ICE} blur(${(1 - k) * 12}px)`, transform: `scale(${1.2 - 0.2 * k + 0.04 * (f / (len * fps))})`, opacity: k }}>{String(p.title ?? "WIRECARD").toUpperCase()}</div>
    <div style={{ fontFamily: FJALLA, fontSize: W * 0.024, letterSpacing: W * 0.004, color: "#9fdcf2", opacity: interpolate(f, [12, 24], [0, 1], clamp) }}>{String(p.subtitle ?? "")}</div>
  </AbsoluteFill>;
};

const Stage: React.FC<P> = () => <AbsoluteFill style={{ background: `radial-gradient(ellipse at 50% 40%, #12263a 0%, ${STEEL} 70%)` }} />;
const Chapter: React.FC<P> = (p) => <Title {...p} title={p.titleCard ? "Wirecard" : p.title} subtitle={p.titleCard ? "The €24 billion company built on money that wasn't there" : ""} />;

export const FILM = {
  components: { Beat, Title, Stage, Chapter, Words: Words as unknown as React.FC<P> },
  roles: { headline: "Words", phrase: "Words", label: "Words", number: "Words", quote: "Words", chapter: "Chapter", stage: "Stage", chart: "Stage", map: "Stage", document: "Stage" },
  transitions: {}, transition: "", transitionFrames: 8,
  fonts: { display: `"${GOTHIC}", sans-serif`, body: `"${FJALLA}", sans-serif` },
  meta: { Beat: { use: "one beat: travelling plates, whips, data/money streaming, words, counter, shove", props: "plates, words, data, money, counter, shove, strobe, whipIn, whipOut, flashes, red" }, Title: { use: "the title card", props: "title, subtitle" } },
};
export { ICE };
