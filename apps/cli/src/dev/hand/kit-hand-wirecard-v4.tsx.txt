// Kit for "Wirecard" v4 (build list 6, 2026-11-25), written from its world bible (apps/cli/src/dev/hand/world-wirecard-v4.md),
// from nothing: no component, colour, typeface or motion of FTX or of Wirecard v1–v3.
// The world: the empire was a stream of transactions that said APPROVED. The film is printed on its own receipts:
// thermal paper printing the figures line by line, a card terminal's green-grey screen, and the FT's salmon-pink paper
// that kept asking where the money was. Courier Prime (the receipt), VT323 (the terminal), Old Standard TT (the FT column).
// No glow, no gradient fill, nothing raining.
import React from "react";
import { AbsoluteFill, Easing, Img, OffthreadVideo, Sequence, interpolate, staticFile, useCurrentFrame, useVideoConfig } from "remotion";
import { Parallax } from "@craft";
import { loadFont as courierF } from "@remotion/google-fonts/CourierPrime";
import { loadFont as vtF } from "@remotion/google-fonts/VT323";
import { loadFont as oldF } from "@remotion/google-fonts/OldStandardTT";
const RECEIPT = courierF("normal", { weights: ["400", "700"], subsets: ["latin"] }).fontFamily;
const LCDF = vtF("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
const COLUMN = oldF("normal", { weights: ["400", "700"], subsets: ["latin"] }).fontFamily;
type P = Record<string, any>;
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const PAPER = "#f3efe6", INK = "#2b2a28", FADED = "#8d8a84", LCD_BG = "#9fb39a", LCD_INK = "#1d2a1c", PINK = "#f2dfcf", PINK_INK = "#33261f", NIGHT = "#0b0d10";
const S = (p: P, k: number) => (p.slots ?? {})[String(k)] as string | undefined;
const src = (s: string) => (/^https?:/.test(s) ? s : staticFile(s));
const ease = Easing.bezier(0.2, 0.7, 0.1, 1);

/** A picture filling the frame (or a box), pushing and drifting the whole time; graded to this world. */
const Plate: React.FC<{ s?: string; trim?: number; from?: number; len: number; push?: [number, number]; pan?: [number, number]; tilt?: [number, number]; grade?: string; box?: [number, number, number, number]; crop?: [number, number] }> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W, height: H } = useVideoConfig();
  if (!p.s) return null;
  const from = p.from ?? 0;
  const k = interpolate(f, [from * fps, (from + p.len) * fps], [0, 1], clamp);
  const [a, b] = p.push ?? [1.08, 1.42], [x0, x1] = p.pan ?? [0.03, -0.03], [r0, r1] = p.tilt ?? [0, 0];
  const isV = /\.(mp4|mov|webm)$/i.test(p.s);
  const st: React.CSSProperties = { position: "absolute", inset: 0, width: "100%", height: "100%", objectFit: "cover", objectPosition: p.crop ? `${p.crop[0] * 100}% ${p.crop[1] * 100}%` : "50% 50%", filter: p.grade ?? "saturate(0.8) contrast(1.18) brightness(0.86)" };
  const pic = isV ? <Sequence from={Math.round(from * fps)} layout="none"><OffthreadVideo src={src(p.s)} muted startFrom={Math.round((p.trim ?? 0) * fps)} style={st} /></Sequence> : <Img src={src(p.s)} style={st} />;
  const box = p.box ? { left: W * p.box[0], top: H * p.box[1], width: W * p.box[2], height: H * p.box[3], overflow: "hidden" as const, boxShadow: "0 30px 60px rgba(0,0,0,.55)" } : { inset: 0 };
  if (f < from * fps - 1) return null;
  return <div style={{ position: "absolute", ...box }}><AbsoluteFill style={{ transform: `scale(${a + (b - a) * k}) translateX(${(x0 + (x1 - x0) * k) * 100}%) rotate(${r0 + (r1 - r0) * k}deg)` }}>{pic}</AbsoluteFill></div>;
};

/** Thermal paper: a strip feeding out of the top of the frame, each line printing on its word (faint at the edges, as
 *  real thermal print is), the strip scrolling up a line each time so the paper keeps moving. */
type Line = { text: string; at: number; size?: number; bold?: boolean; strike?: number; rule?: boolean; align?: "left" | "right" | "center" };
const Receipt: React.FC<{ lines: Line[]; x: number; w: number; tilt?: number; from?: number; until?: number; top?: number }> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W, height: H } = useVideoConfig(); const t = f / fps;
  const from = p.from ?? Math.min(...p.lines.map((l) => l.at)) - 0.5;
  if (t < from) return null;
  const pw = W * p.w, unit = pw * 0.085; // big enough to read at phone size (the channel's words sit at ≈ 6–12 % of frame height)
  const shown = p.lines.filter((l) => t >= l.at - 0.05);
  // the feed: each printed line pushes the paper down out of the slot (top of frame), eased
  const feed = p.lines.reduce((acc, l) => acc + (l.rule ? unit * 0.9 : unit * (l.size ?? 1) * 1.35) * interpolate(t, [l.at - 0.05, l.at + 0.25], [0, 1], { ...clamp, easing: ease }), 0);
  const gone = p.until !== undefined ? interpolate(t, [p.until, p.until + 0.35], [0, 1], { ...clamp, easing: Easing.in(Easing.cubic) }) : 0;
  const drift = (t - from) * pw * 0.012;
  return <div style={{ position: "absolute", left: W * p.x, top: -H * 0.05 + (p.top ?? 0) * H, width: pw, height: feed + unit * 1.6, transform: `rotate(${p.tilt ?? -2}deg) translateY(${drift - gone * H * 1.2}px)`, transformOrigin: "50% 0%",
    background: `linear-gradient(90deg, rgba(0,0,0,.06), rgba(0,0,0,0) 6%, rgba(0,0,0,0) 94%, rgba(0,0,0,.07)), ${PAPER}`, boxShadow: "0 24px 40px rgba(0,0,0,.45), 0 2px 0 rgba(255,255,255,.4) inset",
    clipPath: "polygon(0 0, 100% 0, 100% calc(100% - 10px), 97% 100%, 94% calc(100% - 10px), 91% 100%, 88% calc(100% - 10px), 85% 100%, 82% calc(100% - 10px), 79% 100%, 76% calc(100% - 10px), 73% 100%, 70% calc(100% - 10px), 67% 100%, 64% calc(100% - 10px), 61% 100%, 58% calc(100% - 10px), 55% 100%, 52% calc(100% - 10px), 49% 100%, 46% calc(100% - 10px), 43% 100%, 40% calc(100% - 10px), 37% 100%, 34% calc(100% - 10px), 31% 100%, 28% calc(100% - 10px), 25% 100%, 22% calc(100% - 10px), 19% 100%, 16% calc(100% - 10px), 13% 100%, 10% calc(100% - 10px), 7% 100%, 4% calc(100% - 10px), 0 100%)" }}>
    <div style={{ position: "absolute", left: pw * 0.07, right: pw * 0.07, bottom: unit * 0.9, display: "flex", flexDirection: "column", gap: unit * 0.18 }}>
      {shown.map((l, i) => {
        if (l.rule) return <div key={i} style={{ borderTop: `${Math.max(2, unit * 0.06)}px dashed ${FADED}`, margin: `${unit * 0.25}px 0` }} />;
        const ink = interpolate(t, [l.at - 0.05, l.at + 0.18], [0, 1], clamp);
        const chars = Math.round(l.text.length * interpolate(t, [l.at - 0.05, l.at + 0.22], [0, 1], clamp));
        const strike = l.strike !== undefined ? interpolate(t, [l.strike, l.strike + 0.3], [0, 1], { ...clamp, easing: ease }) : 0;
        const fit = Math.min(unit * (l.size ?? 1), (pw * 0.84) / Math.max(1, l.text.length * 0.62)); // a monospaced face: 0.6 em a character
        return <div key={i} style={{ position: "relative", fontFamily: RECEIPT, fontWeight: l.bold ? 700 : 400, fontSize: fit, lineHeight: 1.05, color: INK, opacity: 0.25 + 0.75 * ink, textAlign: l.align ?? "left", whiteSpace: "pre", letterSpacing: unit * 0.02,
          WebkitMaskImage: "linear-gradient(90deg, rgba(0,0,0,.72), #000 10%, #000 88%, rgba(0,0,0,.6))", filter: "contrast(1.05)" }}>
          {l.text.slice(0, chars)}
          {strike > 0 ? <div style={{ position: "absolute", left: 0, top: "52%", height: Math.max(3, unit * 0.1), width: `${strike * 100}%`, background: INK }} /> : null}
        </div>;
      })}
    </div>
  </div>;
};

/** The card terminal's screen: green-grey LCD, a word per state, a scan refresh each time the word changes.
 *  Placed over a real terminal's screen (box in frame fractions) or alone, large. */
const LCD: React.FC<{ states: Array<{ text: string; at: number; sub?: string }>; box: [number, number, number, number]; rot?: number; skew?: number }> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W, height: H } = useVideoConfig(); const t = f / fps;
  const cur = [...p.states].reverse().find((s) => t >= s.at);
  if (!cur) return null;
  const refresh = interpolate(t, [cur.at, cur.at + 0.18], [0, 1], clamp);
  const bw = W * p.box[2], bh = H * p.box[3];
  return <div style={{ position: "absolute", left: W * p.box[0], top: H * p.box[1], width: bw, height: bh, transform: `rotate(${p.rot ?? 0}deg) skewX(${p.skew ?? 0}deg)`, background: `radial-gradient(ellipse at 50% 40%, #b6c8ae, ${LCD_BG} 70%, #7e9277)`, boxShadow: "inset 0 0 18px rgba(0,0,0,.45)", overflow: "hidden", display: "flex", flexDirection: "column", justifyContent: "center", alignItems: "center" }}>
    <div style={{ fontFamily: LCDF, fontSize: bh * (cur.sub ? 0.5 : 0.66), lineHeight: 0.9, color: LCD_INK, letterSpacing: bh * 0.03, opacity: 0.88, textShadow: "1px 1px 0 rgba(0,0,0,.12)", clipPath: `inset(0 0 ${(1 - refresh) * 100}% 0)` }}>{cur.text}</div>
    {cur.sub ? <div style={{ fontFamily: LCDF, fontSize: bh * 0.26, color: LCD_INK, opacity: 0.8 * refresh }}>{cur.sub}</div> : null}
    <AbsoluteFill style={{ background: "repeating-linear-gradient(0deg, rgba(0,0,0,.05) 0 2px, transparent 2px 4px)" }} />
    <div style={{ position: "absolute", left: 0, right: 0, top: `${refresh * 100}%`, height: bh * 0.08, background: "rgba(20,40,20,.25)", opacity: refresh < 1 ? 1 : 0 }} />
  </div>;
};

/** The FT's salmon-pink newsprint laid over a picture (duotone into the paper's colour, with paper fibre). */
const Sheet: React.FC<{ k: number; box: [number, number, number, number] }> = ({ k, box }) => {
  const { width: W, height: H } = useVideoConfig();
  return <div style={{ position: "absolute", left: W * box[0], top: H * box[1], width: W * box[2], height: H * box[3], opacity: k, transform: `translateX(${(1 - k) * W * 0.1}px) rotate(${box[0] > 0.4 ? 1.2 : -1.2}deg)`, background: "#f4dcc8", boxShadow: "0 30px 60px rgba(0,0,0,.6)" }}>
    <AbsoluteFill style={{ opacity: 0.3, mixBlendMode: "multiply", backgroundImage: "url(\"data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' width='240' height='240'><filter id='n'><feTurbulence type='fractalNoise' baseFrequency='0.9' numOctaves='3'/><feColorMatrix values='0 0 0 0 0.45 0 0 0 0 0.35 0 0 0 0 0.3 0 0 0 0.55 0'/></filter><rect width='240' height='240' filter='url(%23n)'/></svg>\")" }} />
  </div>;
};
const Pink: React.FC<{ k?: number }> = ({ k = 1 }) => <>
  <AbsoluteFill style={{ backdropFilter: "grayscale(1) contrast(1.15)", opacity: k }} />
  <AbsoluteFill style={{ background: "#f4c9ad", mixBlendMode: "multiply", opacity: 0.95 * k }} />
  <AbsoluteFill style={{ background: "#fff4ea", mixBlendMode: "screen", opacity: 0.18 * k }} />
  <AbsoluteFill style={{ opacity: 0.22 * k, mixBlendMode: "multiply", backgroundImage: "url(\"data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' width='240' height='240'><filter id='n'><feTurbulence type='fractalNoise' baseFrequency='0.9' numOctaves='3'/><feColorMatrix values='0 0 0 0 0.45 0 0 0 0 0.35 0 0 0 0 0.3 0 0 0 0.55 0'/></filter><rect width='240' height='240' filter='url(%23n)'/></svg>\")" }} />
</>;

/** A newspaper column on the pink paper: a headline set in the paper's serif, typeset line by line on the voice. */
const Column: React.FC<{ head: Array<{ text: string; at: number }>; x: number; y: number; w: number; size: number; until?: number; kicker?: string }> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W, height: H } = useVideoConfig(); const t = f / fps;
  const first = Math.min(...p.head.map((h) => h.at));
  if (t < first - 0.3) return null;
  const gone = p.until !== undefined ? interpolate(t, [p.until, p.until + 0.3], [1, 0], clamp) : 1;
  return <div style={{ position: "absolute", left: W * p.x, top: H * p.y, width: W * p.w, opacity: gone }}>
    {p.kicker ? <div style={{ fontFamily: COLUMN, fontSize: W * p.size * 0.32, letterSpacing: W * 0.002, color: "#8a3b2a", textTransform: "uppercase", borderBottom: `2px solid ${PINK_INK}`, paddingBottom: 6, marginBottom: 10, opacity: interpolate(t, [first - 0.3, first], [0, 1], clamp) }}>{p.kicker}</div> : null}
    {p.head.map((h, i) => {
      const k = interpolate(t, [h.at - 0.08, h.at + 0.25], [0, 1], { ...clamp, easing: ease });
      return <div key={i} style={{ fontFamily: COLUMN, fontWeight: 700, fontSize: W * p.size, lineHeight: 1.04, color: PINK_INK, clipPath: `inset(0 ${(1 - k) * 100}% 0 0)`, transform: `translateY(${(1 - k) * W * 0.006}px)` }}>{h.text}</div>;
    })}
  </div>;
};

/** A rubber stamp coming down onto the paper: ink pressed unevenly, a little rotation, a jolt on landing. */
const Stamp: React.FC<{ text: string; at: number; x: number; y: number; size: number; rot?: number; colour?: string }> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W, height: H } = useVideoConfig(); const t = f / fps;
  if (t < p.at - 0.12) return null;
  const k = interpolate(t, [p.at - 0.12, p.at], [0, 1], { ...clamp, easing: Easing.in(Easing.quad) });
  const col = p.colour ?? "#b3261e";
  return <div style={{ position: "absolute", left: W * p.x, top: H * p.y, transform: `translate(-50%,-50%) rotate(${p.rot ?? -8}deg) scale(${2.2 - 1.2 * k})`, opacity: k,
    border: `${W * p.size * 0.07}px solid ${col}`, borderRadius: W * p.size * 0.12, padding: `${W * p.size * 0.08}px ${W * p.size * 0.25}px`, fontFamily: RECEIPT, fontWeight: 700, fontSize: W * p.size, color: col, letterSpacing: W * p.size * 0.08,
    mixBlendMode: "multiply", WebkitMaskImage: "radial-gradient(circle at 30% 40%, #000 0 55%, rgba(0,0,0,.55) 70%, #000 90%)" }}>{p.text}</div>;
};

/** A printed label (label-printer tape) for a name: printed out of the edge, then stuck on. */
const Label: React.FC<{ text: string; sub?: string; at: number; x: number; y: number; size: number; until?: number }> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W, height: H } = useVideoConfig(); const t = f / fps;
  if (t < p.at - 0.05) return null;
  const k = interpolate(t, [p.at - 0.05, p.at + 0.35], [0, 1], { ...clamp, easing: ease });
  const gone = p.until !== undefined ? interpolate(t, [p.until, p.until + 0.25], [1, 0], clamp) : 1;
  return <div style={{ position: "absolute", left: W * p.x, top: H * p.y, opacity: gone, clipPath: `inset(0 ${(1 - k) * 100}% 0 0)`, background: PAPER, padding: `${W * p.size * 0.2}px ${W * p.size * 0.45}px`, boxShadow: "0 10px 22px rgba(0,0,0,.45)", transform: "rotate(-1.2deg)" }}>
    <div style={{ fontFamily: RECEIPT, fontWeight: 700, fontSize: W * p.size, color: INK, letterSpacing: W * p.size * 0.06, whiteSpace: "nowrap" }}>{p.text}</div>
    {p.sub ? <div style={{ fontFamily: RECEIPT, fontSize: W * p.size * 0.5, color: FADED, letterSpacing: W * p.size * 0.04 }}>{p.sub}</div> : null}
  </div>;
};

/** The world's transition: the frame tears off along a perforated edge, like a receipt, and is pulled away upward. */
const zig = (y: number, n = 28, d = 1.4) => Array.from({ length: n + 1 }, (_, i) => `${(i / n) * 100}% ${y + (i % 2 ? d : 0)}%`).join(", ");
const TearOut: React.FC<{ len: number; frames?: number; children: React.ReactNode }> = ({ len, frames = 7, children }) => {
  const f = useCurrentFrame(); const { fps, height: H } = useVideoConfig();
  const k = interpolate(f, [len * fps - frames, len * fps], [0, 1], { ...clamp, easing: Easing.in(Easing.cubic) });
  if (k <= 0) return <>{children}</>;
  return <AbsoluteFill style={{ transform: `translateY(${-k * H * 1.05}px) rotate(${-k * 4}deg)`, transformOrigin: "0% 0%", clipPath: `polygon(0 0, 100% 0, ${zig(100 - 1.4).split(", ").reverse().join(", ")})`, boxShadow: "0 20px 40px rgba(0,0,0,.6)" }}>{children}</AbsoluteFill>;
};

/** One beat of the film: real pictures under the world's grade, then the world's own objects on top. */
const Beat: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { fps } = useVideoConfig(); const len = Number(p.len);
  const plates = (p.plates ?? []) as Array<P>;
  // one camera move over the whole beat (pictures and the objects placed on them move together)
  const cam = (p.cam ?? [1, 1]) as [number, number];
  const ck = interpolate(f, [0, len * fps], [0, 1], clamp);
  const camT = `scale(${cam[0] + (cam[1] - cam[0]) * ck})`;
  const body = <AbsoluteFill style={{ background: NIGHT, overflow: "hidden" }}>
   <AbsoluteFill style={{ transform: camT, transformOrigin: `${(p.camAt ?? [0.5, 0.5])[0] * 100}% ${(p.camAt ?? [0.5, 0.5])[1] * 100}%` }}>
    {plates.map((pl, i) => {
      const from = pl.at ?? 0;
      // a later plate tears the earlier one away inside the beat
      const tear = pl.at ? interpolate(f, [from * fps - 8, from * fps], [0, 1], { ...clamp, easing: Easing.in(Easing.cubic) }) : 1;
      return <AbsoluteFill key={i} style={pl.at && tear < 1 ? { clipPath: `polygon(0 ${(1 - tear) * 100}%, 100% ${(1 - tear) * 100}%, 100% 100%, 0 100%)` } : undefined}>
        <Plate s={S(p, pl.slot)} trim={pl.trim} from={Math.max(0, from - 0.3)} len={len - from + 0.3} push={pl.push} pan={pl.pan} tilt={pl.tilt} grade={pl.grade} box={pl.box} crop={pl.crop} />
      </AbsoluteFill>;
    })}
    {p.parallax ? <Parallax bg={S(p, p.parallax.bg)!} cut={S(p, p.parallax.cut)} box={p.parallax.box} aspect={Number(p.parallax.aspect)} keys={p.parallax.keys} near={p.parallax.near} far={p.parallax.far} bgFilter={p.parallax.bgFilter} cutFilter={p.parallax.cutFilter} /> : null}
    {p.pink ? (p.pink.box ? <Sheet box={p.pink.box} k={interpolate(f, [Number(p.pink.from ?? 0) * fps, Number(p.pink.from ?? 0) * fps + 8], [0, 1], { ...clamp, easing: ease })} /> : <Pink k={interpolate(f, [Number(p.pink.from ?? 0) * fps, Number(p.pink.from ?? 0) * fps + 10], [0, 1], clamp)} />) : null}
    {p.lowkey ? <AbsoluteFill style={{ background: "radial-gradient(ellipse at 50% 45%, rgba(0,0,0,0) 30%, rgba(0,0,0,.72) 100%)" }} /> : null}
    {((p.lcd ?? []) as P[]).map((l, i) => <LCD key={i} states={l.states} box={l.box} rot={l.rot} skew={l.skew} />)}
   </AbsoluteFill>
    {((p.columns ?? []) as P[]).map((c, i) => <Column key={i} head={c.head} x={c.x} y={c.y} w={c.w} size={c.size} until={c.until} kicker={c.kicker} />)}
    {((p.receipts ?? []) as P[]).map((r, i) => <Receipt key={i} lines={r.lines} x={r.x} w={r.w} tilt={r.tilt} from={r.from} until={r.until} top={r.top} />)}
    {((p.labels ?? []) as P[]).map((l, i) => <Label key={i} {...(l as never)} />)}
    {((p.stamps ?? []) as P[]).map((s, i) => <Stamp key={i} {...(s as never)} />)}
  </AbsoluteFill>;
  return p.tearOut ? <TearOut len={len}>{body}</TearOut> : body;
};

/** The title: the name printed as a receipt header, the strip torn off the printer. */
const Title: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { fps, width: W } = useVideoConfig(); const len = Number(p.len ?? 2.6);
  const lines: Line[] = [{ text: "* * * * * * * * *", at: 0.1, align: "center" }, { text: String(p.title ?? "WIRECARD").toUpperCase(), at: 0.35, size: 2.6, bold: true, align: "center" }, { text: String(p.subtitle ?? ""), at: 0.9, size: 0.62, align: "center" }, { text: "", at: 1.0, rule: true }, { text: "TOTAL      €0.00", at: 1.35, size: 1.1, bold: true, align: "center" }];
  return <AbsoluteFill style={{ background: NIGHT, opacity: interpolate(f, [len * fps - 8, len * fps], [1, 0], clamp) }}>
    {S(p, 0) ? <Plate s={S(p, 0)} len={len} push={[1.05, 1.2]} pan={[0.02, -0.02]} grade="saturate(1.1) contrast(1.35) brightness(0.42) blur(2px)" /> : null}
    <Receipt lines={lines} x={0.3} w={0.4} tilt={-1} from={0} />
  </AbsoluteFill>;
};
const Stage: React.FC<P> = () => <AbsoluteFill style={{ background: NIGHT }} />;
const Chapter: React.FC<P> = (p) => <Title {...p} title="Wirecard" subtitle="The €24 billion company built on money that wasn't there" />;
const Words: React.FC<P> = (p) => <Receipt lines={(p.lines ?? []) as Line[]} x={Number(p.x ?? 0.32)} w={Number(p.w ?? 0.36)} tilt={Number(p.tilt ?? -2)} until={p.until} />;
