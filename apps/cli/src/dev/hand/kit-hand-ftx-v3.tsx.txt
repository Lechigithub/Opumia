// Kit for "FTX: four days from fine to bankrupt" — v3 art direction "Night exchange" (2026-11-24, rebuilt from
// scratch: the gold-and-brown kit had become a template). An empire that existed on screens and vanished overnight:
// midnight navy, electric cyan accent, signal red only for danger. Four type voices, each kept to one kind of moment:
// wide geometric caps (Unbounded) for headlines, a terminal monospace (IBM Plex Mono) for figures, dates and data, an
// italic serif (Playfair) for human words and quotes, handwriting (Caveat) for the narrator's asides.
import React from "react";
import { AbsoluteFill, Img, interpolate, spring, staticFile, useCurrentFrame, useVideoConfig } from "remotion";
import { loadFont as unbounded } from "@remotion/google-fonts/Unbounded";
import { loadFont as plexMono } from "@remotion/google-fonts/IBMPlexMono";
import { loadFont as playfair } from "@remotion/google-fonts/PlayfairDisplay";
import { loadFont as caveat } from "@remotion/google-fonts/Caveat";
const WIDE = unbounded("normal", { weights: ["700", "900"], subsets: ["latin"] }).fontFamily;
const MONO = plexMono("normal", { weights: ["500", "700"], subsets: ["latin"] }).fontFamily;
const SERIF = playfair("italic", { weights: ["700"], subsets: ["latin"] }).fontFamily;
const HAND = caveat("normal", { weights: ["700"], subsets: ["latin"] }).fontFamily;
const CYAN = "#28e7ff", RED = "#ff2e4d", WHITE = "#ffffff", NAVY = "#070b1a", INK = "#0a1630";
type P = Record<string, any>;
const COL: Record<string, string> = { y: CYAN, w: WHITE, r: RED };
const glow = (c: string) => `0 0 18px ${c}88, 0 0 42px ${c}44, 0 6px 22px rgba(0,0,0,.85)`;
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const useW = (p: P) => { const { width } = useVideoConfig(); return Number(p.boxW ?? width); };
const FONT: Record<string, { family: string; weight: number; upper: boolean; track: number; scale: number }> = {
  wide: { family: WIDE, weight: 900, upper: true, track: -0.01, scale: 0.78 },
  mono: { family: MONO, weight: 700, upper: true, track: 0.02, scale: 0.85 },
  serif: { family: SERIF, weight: 700, upper: false, track: 0, scale: 1 },
  hand: { family: HAND, weight: 700, upper: false, track: 0, scale: 1.15 },
};

/** Words landing one by one on the narration, in one of the film's four voices; key words glow cyan or red. */
const Kinetic: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { fps } = useVideoConfig(); const W = useW(p);
  const words: Array<{ text: string; from: number; c?: string; script?: boolean }> = p.words ?? [];
  const lines: typeof words[] = [[]];
  for (const w of words) { if (w.text === "|") lines.push([]); else lines.at(-1)!.push(w); }
  const v = FONT[String(p.font ?? "wide")] ?? FONT.wide!;
  const longest = Math.max(...lines.map((l) => l.reduce((a, w) => a + w.text.length + 1, 0)));
  const size = Number(p.size ?? Math.min(W * 0.18, (W * 0.95) / (longest * 0.55))) * v.scale;
  const align = String(p.align ?? "center");
  return <div style={{ width: "100%", display: "flex", flexDirection: "column", alignItems: align === "left" ? "flex-start" : align === "right" ? "flex-end" : "center", gap: size * 0.06, transform: `scale(${1 + f * 0.0014})` }}>
    {lines.map((l, li) => <div key={li} style={{ display: "flex", gap: size * 0.28, alignItems: "baseline", flexWrap: "wrap", justifyContent: align === "left" ? "flex-start" : "center" }}>{l.map((w, i) => {
      const k = spring({ frame: f - w.from, fps, config: { damping: 15, stiffness: 200, mass: 0.6 } });
      const c = COL[w.c ?? "w"]!;
      return <span key={i} style={{ fontFamily: w.script ? HAND : v.family, fontWeight: w.script ? 700 : v.weight, fontSize: w.script ? size * 1.15 : size, lineHeight: 1.02, textTransform: v.upper && !w.script ? "uppercase" : "none", letterSpacing: size * v.track, color: c, textShadow: w.c ? glow(c) : "0 6px 24px rgba(0,0,0,.85)", opacity: f >= w.from ? Math.min(1, k * 1.8) : 0, transform: `translateY(${(1 - k) * size * 0.35}px)`, filter: `blur(${(1 - Math.min(1, k)) * 5}px)`, display: "inline-block" }}>{w.text}</span>;
    })}</div>)}
  </div>;
};

/** A figure ticking up like a trading screen: monospace digits with a cyan (or red) glow, its label above. */
const Counter: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p); const { height: FH } = useVideoConfig();
  const to = Number(p.to ?? 0), dec = Number(p.decimals ?? 0), dur = Number(p.frames ?? 24);
  const v = interpolate(f, [0, dur], [Number(p.from ?? to * 0.2), to], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) });
  const txt = `${p.prefix ?? ""}${v.toFixed(dec)}${p.suffix ?? ""}`, fin = `${p.prefix ?? ""}${to.toFixed(dec)}${p.suffix ?? ""}`;
  const size = Number(p.size ?? Math.min((W * 0.95) / (fin.length * 0.62), FH * 0.3));
  const c = p.fill === "red" ? RED : CYAN;
  return <div style={{ width: "100%", textAlign: "center", opacity: interpolate(f, [0, 4], [0, 1], clamp) }}>
    {p.label ? <div style={{ fontFamily: MONO, fontWeight: 500, fontSize: size * 0.17, color: c, letterSpacing: size * 0.04, textTransform: "uppercase", marginBottom: size * 0.04, opacity: interpolate(f, [4, 10], [0, 1], clamp) }}>{"▍ " + String(p.label)}</div> : null}
    <div style={{ fontFamily: MONO, fontWeight: 700, fontSize: size, lineHeight: 1, color: WHITE, textShadow: glow(c), letterSpacing: -size * 0.02 }}>{txt}</div>
  </div>;
};

/** Danger: a red error banner slammed across the frame. */
const Stamp: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const k = interpolate(f, [0, 5], [1.6, 1], { ...clamp, easing: (t) => t * t });
  const size = Number(p.size ?? W * 0.14);
  const jitter = f < 10 ? Math.sin(f * 7) * 6 : 0;
  return <div style={{ display: "inline-block", transform: `scale(${k}) translateX(${jitter}px) rotate(${p.rot ?? -3}deg)`, background: RED, color: WHITE, fontFamily: WIDE, fontWeight: 900, fontSize: size, lineHeight: 1, padding: `${size * 0.12}px ${size * 0.3}px`, letterSpacing: size * 0.02, textTransform: "uppercase", boxShadow: `0 0 40px ${RED}aa` }}>{String(p.text ?? "")}</div>;
};

/** A real document or post on a lit screen floating in the dark: cyan edge, slight 3D tilt, a highlight sweeping. */
const Card: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { fps } = useVideoConfig(); const W = useW(p);
  const src = p.slots?.[String(p.slot ?? 0)];
  const k = spring({ frame: f, fps, config: { damping: 16, stiffness: 110 } });
  const hl = p.hl as { x: number; y: number; w: number; h: number; from?: number; to?: number } | undefined;
  const sweep = hl ? interpolate(f, [hl.from ?? 12, hl.to ?? 30], [0, 1], clamp) : 0;
  const crop = p.crop as { y0: number; y1: number } | undefined;
  return <div style={{ width: W, perspective: W * 2.4 }}>
    <div style={{ transform: `rotateY(${(p.rot ?? -8) * (1 - 0.3 * k)}deg) rotateX(4deg) translateY(${(1 - k) * 120}px) scale(${1 + f * 0.003})`, opacity: Math.min(1, k * 1.6), position: "relative", background: "#0d1426", padding: W * 0.012, borderRadius: W * 0.012, border: `2px solid ${CYAN}`, boxShadow: `0 0 50px ${CYAN}55, 0 30px 80px rgba(0,0,0,.9)` }}>
      {src ? (crop ? <div style={{ position: "relative", width: "100%", aspectRatio: `1 / ${(crop.y1 - crop.y0) * Number(p.aspect ?? 0.5)}`, overflow: "hidden", borderRadius: W * 0.006 }}><Img src={staticFile(src)} style={{ position: "absolute", left: 0, width: "100%", top: `${(-crop.y0 / (crop.y1 - crop.y0)) * 100}%` }} /></div> : <Img src={staticFile(src)} style={{ width: "100%", display: "block", borderRadius: W * 0.006 }} />) : null}
      {hl ? <div style={{ position: "absolute", left: `calc(${hl.x * 100}% + ${W * 0.012}px)`, top: `calc(${hl.y * 100}% + ${W * 0.012}px)`, width: `${hl.w * 100 * sweep}%`, height: `${hl.h * 100}%`, background: CYAN, mixBlendMode: "multiply", opacity: 0.75 }} /> : null}
    </div>
  </div>;
};

/** A share of a whole as a ring: the part in red, the rest a thin cyan line, the figure in monospace. */
const Donut: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const share = Number(p.share ?? 0.4); const r = W * 0.38; const C = 2 * Math.PI * r;
  const a = interpolate(f, [Number(p.from ?? 4), Number(p.frames ?? 26)], [0, share], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) });
  return <div style={{ width: W, height: W, position: "relative", opacity: interpolate(f, [0, 6], [0, 1], clamp) }}>
    <svg width={W} height={W} viewBox={`0 0 ${W} ${W}`}>
      <circle cx={W / 2} cy={W / 2} r={r} fill="none" stroke={CYAN} strokeOpacity={0.5} strokeWidth={W * 0.02} />
      <circle cx={W / 2} cy={W / 2} r={r} fill="none" stroke={RED} strokeWidth={W * 0.1} strokeDasharray={`${C * a} ${C}`} transform={`rotate(-90 ${W / 2} ${W / 2})`} style={{ filter: `drop-shadow(0 0 20px ${RED})` }} />
    </svg>
    <div style={{ position: "absolute", inset: 0, display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center" }}>
      <div style={{ fontFamily: MONO, fontWeight: 700, fontSize: W * 0.24, color: WHITE, textShadow: glow(RED), lineHeight: 1 }}>{Math.round(a * 100)}%</div>
      <div style={{ fontFamily: MONO, fontWeight: 500, fontSize: W * 0.08, color: CYAN, letterSpacing: W * 0.01 }}>{String(p.label ?? "")}</div>
    </div>
  </div>;
};

/** A human voice: the words in italic serif, the speaker in monospace below. */
const Quote: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const words: Array<{ text: string; from: number; c?: string }> = p.words ?? [];
  const size = Number(p.size ?? W * 0.07);
  return <div style={{ width: "100%" }}>
    <div style={{ display: "flex", flexWrap: "wrap", gap: `${size * 0.1}px ${size * 0.26}px` }}>{words.map((w, i) => { const a = interpolate(f, [w.from, w.from + 5], [0, 1], clamp); return <span key={i} style={{ fontFamily: SERIF, fontWeight: 700, fontSize: size, color: COL[w.c ?? "w"], textShadow: w.c ? glow(COL[w.c]!) : "0 6px 24px rgba(0,0,0,.85)", opacity: a, transform: `translateY(${(1 - a) * 14}px)` }}>{w.text}</span>; })}</div>
    {p.by ? <div style={{ fontFamily: MONO, fontSize: size * 0.3, color: CYAN, marginTop: size * 0.3, letterSpacing: size * 0.02, opacity: interpolate(f, [Number(p.byFrom ?? 20), Number(p.byFrom ?? 20) + 8], [0, 1], clamp) }}>{"— " + String(p.by)}</div> : null}
  </div>;
};

/** A HUD tag: a cyan bracket and monospace caps, typed on. */
const Label: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const t = String(p.text ?? ""); const n = Math.round(interpolate(f, [0, Math.max(6, t.length * 1.2)], [0, t.length], clamp));
  const fs = Number(p.size ?? W * 0.1) * 0.62;
  return <div style={{ display: "inline-flex", alignItems: "center", gap: fs * 0.4, padding: `${fs * 0.25}px ${fs * 0.5}px`, border: `2px solid ${CYAN}`, background: "rgba(7,11,26,.72)", boxShadow: `0 0 24px ${CYAN}55` }}>
    <div style={{ width: fs * 0.32, height: fs * 0.32, background: RED, borderRadius: "50%", opacity: Math.floor(f / 8) % 2 ? 1 : 0.35 }} />
    <div style={{ fontFamily: MONO, fontWeight: 700, fontSize: fs, color: WHITE, letterSpacing: fs * 0.06, textTransform: "uppercase" }}>{t.slice(0, n)}</div>
  </div>;
};

/** A name beside its person: wide caps sized to the box, the role in monospace cyan above. */
const Name: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const lines = String(p.text ?? "").split("|").map((x) => x.trim());
  const size = Number(p.size ?? (W * 0.96) / (Math.max(...lines.map((l) => l.length)) * 0.86));
  const align = String(p.align ?? "left");
  return <div style={{ width: "100%", textAlign: align as "left" }}>
    {p.role ? <div style={{ fontFamily: MONO, fontWeight: 500, fontSize: size * 0.24, color: CYAN, letterSpacing: size * 0.03, textTransform: "uppercase", textShadow: glow(CYAN), opacity: interpolate(f, [8, 16], [0, 1], clamp) }}>{"▍ " + String(p.role)}</div> : null}
    {lines.map((l, i) => { const a = interpolate(f, [i * 4, i * 4 + 9], [0, 1], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) }); return <div key={i} style={{ fontFamily: WIDE, fontWeight: 900, fontSize: size, lineHeight: 1, color: WHITE, textTransform: "uppercase", textShadow: "0 6px 30px rgba(0,0,0,.9)", opacity: a, transform: `translateX(${(1 - a) * (align === "right" ? 50 : -50)}px)` }}>{l}</div>; })}
  </div>;
};

const Arrow: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p); const H = Number(p.boxH ?? W);
  const k = interpolate(f, [0, 16], [0, 1], clamp);
  return <svg width={W} height={H} style={{ overflow: "visible" }}><path d={`M ${W * 0.1} ${H * 0.1} L ${W * 0.45} ${H * 0.5} L ${W * 0.6} ${H * 0.35} L ${W * 0.9} ${H * 0.85}`} fill="none" stroke={RED} strokeWidth={W * 0.05} pathLength={1} strokeDasharray={1} strokeDashoffset={1 - k} style={{ filter: `drop-shadow(0 0 14px ${RED})` }} /></svg>;
};

/** The stage behind shots with no footage: a navy field with a faint cyan grid drifting and scanlines. */
const Stage: React.FC<P> = () => {
  const f = useCurrentFrame(); const { width: W } = useVideoConfig();
  return <AbsoluteFill style={{ background: `radial-gradient(ellipse at 50% 40%, #13306b 0%, ${NAVY} 70%)` }}>
    <AbsoluteFill style={{ backgroundImage: `linear-gradient(${CYAN}14 1px, transparent 1px), linear-gradient(90deg, ${CYAN}14 1px, transparent 1px)`, backgroundSize: `${W / 22}px ${W / 22}px`, backgroundPosition: `${f * 0.6}px ${f * 0.3}px` }} />
    <AbsoluteFill style={{ background: "repeating-linear-gradient(0deg, rgba(0,0,0,0) 0, rgba(0,0,0,0) 2px, rgba(0,0,0,.18) 3px)" }} />
  </AbsoluteFill>;
};

const Chapter: React.FC<P> = (p) => { const f = useCurrentFrame(); const { width: W } = useVideoConfig(); return <AbsoluteFill style={{ background: NAVY, justifyContent: "center", alignItems: "center" }}><div style={{ fontFamily: WIDE, fontWeight: 900, fontSize: W * 0.09, color: WHITE, opacity: interpolate(f, [0, 10], [0, 1], clamp) }}>{String(p.title ?? "")}</div></AbsoluteFill>; };
const Chart: React.FC<P> = (p) => <AbsoluteFill><Stage /><div style={{ position: "absolute", left: "10%", top: "10%", fontFamily: WIDE, fontSize: 70, color: WHITE }}>{String(p.title ?? "")}</div></AbsoluteFill>;
const MapC: React.FC<P> = (p) => <AbsoluteFill><Stage /><div style={{ position: "absolute", left: "10%", top: "10%", fontFamily: WIDE, fontSize: 70, color: WHITE }}>{String(p.title ?? "")}</div></AbsoluteFill>;
const Doc: React.FC<P> = (p) => <AbsoluteFill><Stage /><div style={{ position: "absolute", left: "15%", right: "15%", top: "15%", fontFamily: MONO, fontSize: 40, color: WHITE }}>{String(p.headline ?? "")}</div></AbsoluteFill>;

/** The cut: a cyan data glitch — horizontal bars tearing across, then clearing. */
const DataWipe: React.FC<{ progress: number }> = ({ progress }) => {
  const a = progress < 0.5 ? progress * 2 : (1 - progress) * 2;
  return <AbsoluteFill style={{ mixBlendMode: "screen" }}>{Array.from({ length: 9 }, (_, i) => <div key={i} style={{ position: "absolute", left: 0, right: 0, top: `${i * 11 + ((progress * 37 * (i + 1)) % 7)}%`, height: `${1 + (i % 3)}%`, background: CYAN, opacity: a * (0.12 + (i % 3) * 0.1), transform: `translateX(${(i % 2 ? 1 : -1) * (1 - a) * 30}%)` }} />)}</AbsoluteFill>;
};
