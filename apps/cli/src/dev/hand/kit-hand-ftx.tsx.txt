// Hand-made kit for "FTX: four days from fine to bankrupt" (written in the session, no model).
// Concept: "The receipt" — gold light on black money; every claim lands as huge condensed caps word by word, the
// documents themselves are taped down and highlighted, and each cut is a burst of gold light.
import React from "react";
import { AbsoluteFill, Img, interpolate, spring, staticFile, useCurrentFrame, useVideoConfig } from "remotion";
import { loadFont as anton } from "@remotion/google-fonts/Anton";
import { loadFont as oswald } from "@remotion/google-fonts/Oswald";
import { loadFont as marker } from "@remotion/google-fonts/PermanentMarker";
import { loadFont as montserrat } from "@remotion/google-fonts/Montserrat";
const D = anton("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
const B = oswald("normal", { weights: ["300", "500", "700"], subsets: ["latin"] }).fontFamily;
const X = montserrat("normal", { weights: ["800"], subsets: ["latin"] }).fontFamily;
const S = marker("normal", { weights: ["400"], subsets: ["latin"] }).fontFamily;
const GOLD = "#ffcc00", WHITE = "#ffffff", RED = "#ff2b2b", INK = "#0b0806", PAPER = "#f4efe4";
type P = Record<string, any>;
const COL: Record<string, string> = { y: GOLD, w: WHITE, r: RED };
const heavy = "0 6px 0 rgba(0,0,0,.55), 0 10px 40px rgba(0,0,0,.85)";
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const useW = (p: P) => { const { width } = useVideoConfig(); return Number(p.boxW ?? width); };

/** Words landing one by one on the narration: each pops from 130 % with a blur, then settles; lines wrap at `|`. */
const Kinetic: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { fps } = useVideoConfig(); const W = useW(p);
  const words: Array<{ text: string; from: number; c?: string; script?: boolean }> = p.words ?? [];
  const lines: typeof words[] = [[]];
  for (const w of words) { if (w.text === "|") lines.push([]); else lines.at(-1)!.push(w); }
  const longest = Math.max(...lines.map((l) => l.reduce((a, w) => a + w.text.length + 1, 0)));
  const size = Number(p.size ?? Math.min(W * 0.2, (W * 0.95) / (longest * 0.5)));
  const align = String(p.align ?? "center");
  // the second voice (v13): wide heavy caps. The channel's lettering is heavy (median stroke 0.14 of the line height,
  // 54–85 % heavy lines) and mixes narrow heavy caps with wide rounded ones; light type is rare (type-band.py)
  const thin = p.font === "thin", wide = p.font === "wide";
  const live = 1 + f * 0.0016;
  return <div style={{ width: "100%", display: "flex", flexDirection: "column", alignItems: align === "left" ? "flex-start" : "center", gap: size * 0.02, transform: `scale(${live}) translateY(${Math.sin(f / 18) * 4}px)` }}>
    {lines.map((l, li) => <div key={li} style={{ display: "flex", gap: size * 0.22, alignItems: "baseline" }}>{l.map((w, i) => {
      const k = spring({ frame: f - w.from, fps, config: { damping: 13, stiffness: 220, mass: 0.6 } });
      const vis = f >= w.from ? 1 : 0;
      const sc = interpolate(k, [0, 1], [1.35, 1]);
      return <span key={i} style={{ fontFamily: w.script ? S : wide ? X : thin ? B : D, fontWeight: wide ? 800 : thin ? 300 : undefined, fontSize: w.script ? size * 0.82 : wide ? size * 0.92 : size, lineHeight: 1, textTransform: w.script || thin ? "none" : "uppercase", color: COL[w.c ?? "w"], textShadow: thin ? "0 4px 30px rgba(0,0,0,.9)" : heavy, opacity: vis * Math.min(1, k * 2), transform: `scale(${sc}) rotate(${w.script ? -4 : 0}deg)`, filter: `blur(${(1 - Math.min(1, k)) * 6}px)`, display: "inline-block", letterSpacing: w.script ? 0 : thin ? size * 0.06 : wide ? -size * 0.01 : size * 0.01 }}>{w.text}</span>;
    })}</div>)}
  </div>;
};

/** A figure that counts up to its value and fills the frame's width (the channel's figures span most of the frame), in
 *  metallic gold (or red) with a hot glow; its label in light condensed type above it. */
const Counter: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const to = Number(p.to ?? 0), dec = Number(p.decimals ?? 0), dur = Number(p.frames ?? 24);
  const v = interpolate(f, [0, dur], [Number(p.from ?? to * 0.3), to], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) });
  const txt = `${p.prefix ?? ""}${v.toFixed(dec)}${p.suffix ?? ""}`;
  const fin = `${p.prefix ?? ""}${to.toFixed(dec)}${p.suffix ?? ""}`;
  const { height: FH } = useVideoConfig();
  // Anton figures are ≈ 0.55 em wide; never taller than 42 % of the frame (the channel's big figures span the width,
  // not the height)
  const size = Number(p.size ?? Math.min((W * 0.95) / (fin.length * 0.55), FH * 0.42));
  const pop = interpolate(f, [dur, dur + 4, dur + 9], [1, 1.06, 1], clamp);
  const red = p.fill === "red";
  const metal = red ? "linear-gradient(180deg,#ffb3a8 0%,#ff2b2b 42%,#8a0c0c 58%,#ff5a4a 100%)" : "linear-gradient(180deg,#fff6c8 0%,#ffd23a 38%,#a86e00 55%,#ffcc00 78%,#fff0a0 100%)";
  const glow = red ? "rgba(255,43,43,.55)" : "rgba(255,190,40,.5)";
  return <div style={{ width: "100%", textAlign: "center", opacity: interpolate(f, [0, 4], [0, 1], clamp), transform: `scale(${1 + f * 0.0012})` }}>
    {p.label ? <div style={{ fontFamily: X, fontWeight: 800, textTransform: "uppercase", fontSize: size * 0.2, color: WHITE, letterSpacing: size * 0.02, textShadow: "0 4px 24px rgba(0,0,0,.9)", marginBottom: -size * 0.04, opacity: interpolate(f, [6, 12], [0, 1], clamp) }}>{String(p.label)}</div> : null}
    <div style={{ position: "relative", fontFamily: D, fontSize: size, lineHeight: 1, transform: `scale(${pop})` }}>
      <div style={{ position: "absolute", inset: 0, color: "transparent", textShadow: `0 0 ${size * 0.25}px ${glow}, 0 ${size * 0.04}px ${size * 0.08}px rgba(0,0,0,.8)` }}>{txt}</div>
      <div style={{ position: "relative", background: metal, WebkitBackgroundClip: "text", backgroundClip: "text", color: "transparent", WebkitTextStroke: `${Math.max(1, size * 0.006)}px rgba(255,255,255,.35)` }}>{txt}</div>
    </div>
  </div>;
};

/** A red rubber stamp slammed onto the frame. */
const Stamp: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const k = interpolate(f, [0, 5], [2.4, 1], { ...clamp, easing: (t) => t * t });
  const shake = f >= 5 && f < 11 ? Math.sin(f * 3) * 6 : 0;
  const size = Number(p.size ?? W * 0.16);
  return <div style={{ display: "inline-block", transform: `rotate(${p.rot ?? -9}deg) scale(${k}) translate(${shake}px, 0)`, opacity: f >= 0 ? interpolate(f, [0, 3], [0, 0.92], clamp) : 0, border: `${size * 0.09}px solid ${RED}`, padding: `${size * 0.05}px ${size * 0.2}px`, color: RED, fontFamily: D, fontSize: size, lineHeight: 1, letterSpacing: size * 0.04, textTransform: "uppercase", mixBlendMode: "screen", filter: "drop-shadow(0 0 12px rgba(255,43,43,.5))" }}>{String(p.text ?? "")}</div>;
};

/** A real document (picture slot) taped down on the frame, with a highlighter sweeping a region of it. */
const Card: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { fps } = useVideoConfig(); const W = useW(p);
  const src = p.slots?.[String(p.slot ?? 0)];
  const k = spring({ frame: f, fps, config: { damping: 14, stiffness: 120 } });
  const hl = p.hl as { x: number; y: number; w: number; h: number; from?: number; to?: number } | undefined;
  const sweep = hl ? interpolate(f, [hl.from ?? 12, hl.to ?? 30], [0, 1], clamp) : 0;
  const push = 1 + f * 0.0045; // keeps pushing for as long as it is on screen
  return <div style={{ width: W, transform: `translate(${Math.sin(f / 20) * 10}px, ${(1 - k) * 160}px) rotate(${(p.rot ?? -2) * k + Math.sin(f / 26) * 0.8}deg) scale(${push})`, opacity: Math.min(1, k * 1.6), position: "relative", background: PAPER, padding: W * 0.018, boxShadow: "0 30px 70px rgba(0,0,0,.8)" }}>
    {src ? <Img src={staticFile(src)} style={{ width: "100%", display: "block" }} /> : null}
    {hl ? <div style={{ position: "absolute", left: `calc(${hl.x * 100}% + ${W * 0.018}px)`, top: `calc(${hl.y * 100}% + ${W * 0.018}px)`, width: `${hl.w * 100 * sweep}%`, height: `${hl.h * 100}%`, background: GOLD, mixBlendMode: "multiply", opacity: 0.85 }} /> : null}
    {[[-0.03, -0.035, -8], [0.86, -0.04, 7]].map(([x, y, r], i) => <div key={i} style={{ position: "absolute", left: `${x * 100}%`, top: `${y * 100}%`, width: W * 0.16, height: W * 0.045, background: "rgba(235,225,190,.82)", transform: `rotate(${r}deg)`, boxShadow: "0 2px 6px rgba(0,0,0,.3)" }} />)}
  </div>;
};

/** The share of a whole as a donut: the part in red with its figure in the middle. */
const Donut: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const share = Number(p.share ?? 0.4); const r = W * 0.36; const C = 2 * Math.PI * r;
  const a = interpolate(f, [6, 30], [0, share], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) });
  const k = interpolate(f, [0, 8], [0, 1], clamp);
  return <div style={{ width: W, height: W, position: "relative", opacity: k, transform: `scale(${0.8 + 0.2 * k})` }}>
    <svg width={W} height={W} viewBox={`0 0 ${W} ${W}`}>
      <circle cx={W / 2} cy={W / 2} r={r} fill="none" stroke="rgba(255,204,0,.9)" strokeWidth={W * 0.11} />
      <circle cx={W / 2} cy={W / 2} r={r} fill="none" stroke={RED} strokeWidth={W * 0.13} strokeDasharray={`${C * a} ${C}`} transform={`rotate(-90 ${W / 2} ${W / 2})`} style={{ filter: "drop-shadow(0 0 18px rgba(255,43,43,.7))" }} />
    </svg>
    <div style={{ position: "absolute", inset: 0, display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center" }}>
      <div style={{ fontFamily: D, fontSize: W * 0.26, color: WHITE, textShadow: heavy, lineHeight: 1 }}>{Math.round(a * 100)}%</div>
      <div style={{ fontFamily: D, fontSize: W * 0.1, color: RED, textShadow: heavy, lineHeight: 1 }}>{String(p.label ?? "")}</div>
    </div>
  </div>;
};

/** A quote revealed word by word in white caps, key words gold, the speaker in the script hand. */
const Quote: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const words: Array<{ text: string; from: number; c?: string }> = p.words ?? [];
  const size = Number(p.size ?? W * 0.062);
  return <div style={{ width: "100%" }}>
    <div style={{ fontFamily: D, fontSize: size * 2.6, color: GOLD, lineHeight: 0.6, textShadow: heavy, opacity: interpolate(f, [0, 6], [0, 1], clamp) }}>“</div>
    <div style={{ display: "flex", flexWrap: "wrap", gap: `${size * 0.1}px ${size * 0.28}px` }}>{words.map((w, i) => {
      const a = interpolate(f, [w.from, w.from + 4], [0, 1], clamp);
      return <span key={i} style={{ fontFamily: D, fontSize: size, lineHeight: 1.1, textTransform: "uppercase", color: COL[w.c ?? "w"], textShadow: heavy, opacity: a, transform: `translateY(${(1 - a) * 18}px)` }}>{w.text}</span>;
    })}</div>
    {p.by ? <div style={{ fontFamily: S, fontSize: size * 0.8, color: GOLD, marginTop: size * 0.5, textShadow: heavy, opacity: interpolate(f, [Number(p.byFrom ?? 20), Number(p.byFrom ?? 20) + 8], [0, 1], clamp), transform: "rotate(-3deg)" }}>— {String(p.by)}</div> : null}
  </div>;
};

/** A name tag: a gold bar wipes in, the name in condensed caps, the role under it. */
const Label: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const k = interpolate(f, [0, 8], [0, 1], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) });
  const fs = Number(p.size ?? W * 0.13);
  return <div style={{ display: "inline-block", clipPath: `inset(0 ${(1 - k) * 100}% 0 0)` }}>
    <div style={{ background: GOLD, color: INK, fontFamily: D, fontSize: fs, lineHeight: 1, padding: `${fs * 0.12}px ${fs * 0.3}px`, textTransform: "uppercase" }}>{String(p.text ?? "")}</div>
    {p.sub ? <div style={{ background: INK, color: WHITE, fontFamily: B, fontWeight: 700, fontSize: fs * 0.42, padding: `${fs * 0.08}px ${fs * 0.3}px`, letterSpacing: 2, textTransform: "uppercase", display: "inline-block" }}>{String(p.sub)}</div> : null}
  </div>;
};

/** A name set huge beside its person (the channel never uses a lower-third box): condensed caps sized to the box's
 *  width, the role in light type above it. Lines split at "|". */
const Name: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p);
  const lines = String(p.text ?? "").split("|").map((x) => x.trim());
  const longest = Math.max(...lines.map((l) => l.length));
  const size = Number(p.size ?? (W * 0.98) / (longest * 0.43));
  const align = String(p.align ?? "left");
  return <div style={{ width: "100%", textAlign: align as "left", transform: `scale(${1 + f * 0.0012})`, transformOrigin: align === "right" ? "right center" : "left center" }}>
    {p.role ? <div style={{ fontFamily: X, fontWeight: 800, textTransform: "uppercase", fontSize: size * 0.24, color: COL[String(p.roleC ?? "y")], letterSpacing: size * 0.02, textShadow: "0 2px 0 rgba(0,0,0,.9), 0 0 18px rgba(0,0,0,.95), 0 0 40px rgba(0,0,0,.8)", opacity: interpolate(f, [8, 16], [0, 1], clamp) }}>{String(p.role)}</div> : null}
    {lines.map((l, i) => { const a = interpolate(f, [i * 4, i * 4 + 8], [0, 1], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 3) }); return <div key={i} style={{ fontFamily: D, fontSize: size, lineHeight: 0.95, color: WHITE, textTransform: "uppercase", textShadow: heavy, opacity: a, transform: `translateX(${(1 - a) * (align === "right" ? 60 : -60)}px)` }}>{l}</div>; })}
  </div>;
};

/** A thick arrow drawing itself down (a fall) or up. */
const Arrow: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const W = useW(p); const H = Number(p.boxH ?? W);
  const k = interpolate(f, [0, 16], [0, 1], { ...clamp, easing: (t) => 1 - Math.pow(1 - t, 2) });
  const down = p.dir !== "up";
  const pts = down ? `M ${W * 0.1} ${H * 0.1} L ${W * 0.4} ${H * 0.45} L ${W * 0.55} ${H * 0.32} L ${W * 0.86} ${H * 0.82}` : `M ${W * 0.1} ${H * 0.9} L ${W * 0.86} ${H * 0.15}`;
  return <svg width={W} height={H} style={{ overflow: "visible", filter: "drop-shadow(0 8px 20px rgba(0,0,0,.8))" }}>
    <path d={pts} fill="none" stroke={p.c === "y" ? GOLD : RED} strokeWidth={W * 0.06} strokeLinecap="square" strokeLinejoin="miter" pathLength={1} strokeDasharray={1} strokeDashoffset={1 - k} />
    {k > 0.95 ? <polygon points={down ? `${W * 0.95},${H * 0.95} ${W * 0.7},${H * 0.86} ${W * 0.9},${H * 0.66}` : `${W * 0.95},${H * 0.05} ${W * 0.7},${H * 0.1} ${W * 0.88},${H * 0.3}`} fill={p.c === "y" ? GOLD : RED} /> : null}
  </svg>;
};

/** Gold light streaks drifting across black: the stage behind shots with no footage. */
const Stage: React.FC<P> = () => {
  const f = useCurrentFrame();
  return <AbsoluteFill style={{ background: `radial-gradient(ellipse at 50% 55%, #2a1a06 0%, ${INK} 70%)` }}>
    {[0, 1, 2].map((i) => <div key={i} style={{ position: "absolute", left: `${-40 + ((f * (0.6 + i * 0.25) + i * 45) % 160)}%`, top: `${18 + i * 27}%`, width: "60%", height: 3 + i * 2, background: `linear-gradient(90deg, transparent, rgba(255,204,0,${0.35 - i * 0.08}), transparent)`, filter: "blur(2px)", transform: "rotate(-8deg)" }} />)}
  </AbsoluteFill>;
};

/** Chapter / title lock-up on black. */
const Chapter: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { width: W } = useVideoConfig();
  const k = interpolate(f, [0, 10], [0, 1], clamp);
  return <AbsoluteFill style={{ background: INK, justifyContent: "center", alignItems: "center" }}>
    <div style={{ fontFamily: D, fontSize: W * 0.12, color: WHITE, opacity: k, letterSpacing: W * 0.004 * (1 - k) * 10, textTransform: "uppercase" }}>{String(p.title ?? "")}</div>
    {p.subtitle ? <div style={{ fontFamily: S, fontSize: W * 0.04, color: GOLD, opacity: interpolate(f, [8, 18], [0, 1], clamp) }}>{String(p.subtitle)}</div> : null}
  </AbsoluteFill>;
};

/** A plain line chart drawn from verified values (role contract); gold line, red last point. */
const Chart: React.FC<P> = (p) => {
  const f = useCurrentFrame(); const { width: W, height: H } = useVideoConfig();
  const vals: number[] = p.values ?? []; const mn = Math.min(...vals), mx = Math.max(...vals);
  const X = (i: number) => W * 0.12 + (i * W * 0.76) / Math.max(1, vals.length - 1);
  const Y = (v: number) => H * 0.8 - ((v - mn) / Math.max(1e-9, mx - mn)) * H * 0.5;
  const pr = interpolate(f, [0, 30], [0, 1], clamp);
  return <AbsoluteFill><Stage /><svg width={W} height={H}><path d={vals.map((v, i) => `${i ? "L" : "M"}${X(i)} ${Y(v)}`).join(" ")} fill="none" stroke={GOLD} strokeWidth={W * 0.008} pathLength={1} strokeDasharray={1} strokeDashoffset={1 - pr} /></svg>
    <div style={{ position: "absolute", left: "12%", top: "8%", fontFamily: D, fontSize: W * 0.06, color: WHITE, textTransform: "uppercase" }}>{String(p.title ?? "")}</div></AbsoluteFill>;
};
const MapC: React.FC<P> = (p) => <AbsoluteFill><Stage /><div style={{ position: "absolute", left: "10%", top: "10%", fontFamily: D, fontSize: 80, color: WHITE }}>{String(p.title ?? "")}</div></AbsoluteFill>;
const Doc: React.FC<P> = (p) => <AbsoluteFill><Stage /><div style={{ position: "absolute", left: "15%", right: "15%", top: "15%", background: PAPER, padding: 60, fontFamily: B, fontSize: 48, color: INK }}>{String(p.headline ?? "")}<br />{String(p.body ?? "")}</div></AbsoluteFill>;

/** The cut: a burst of gold light that blooms and burns out. */
const GoldFlash: React.FC<{ progress: number }> = ({ progress }) => {
  const a = progress < 0.5 ? progress * 2 : (1 - progress) * 2;
  return <AbsoluteFill style={{ background: `radial-gradient(circle at 50% 50%, rgba(255,250,230,${a}) 0%, rgba(255,190,40,${a * 0.85}) 35%, rgba(255,120,0,${a * 0.4}) 70%, transparent 100%)`, mixBlendMode: "screen" }} />;
};
